\section{命题逻辑}

\subsection{命题与联接词}

\begin{definition}{命题}
	命题 (Proposition) 是逻辑的基本成分, 命题是一个或真或假的陈述语句, 但不可能即真又假.我们习惯用小写字母 $p, q, r, s, \cdots$ 表示命题.如果一个命题为真, 用 T 表示, 反之用 F 表示.
	
	特别的, 对于无法拆分成更简单语句的命题, 我们称之为简单命题或原子命题, 而称由联接词和简单命题构成的命题为复合命题.
\end{definition}


\begin{definition}{命题常项与命题变项}
对于简单命题来说, 其真值是确定的, 因而又称命题常项 (Propositional constant), 而对于一些含有变量的陈述句, 虽然不是命题, 但只要给定变量的值就可以得到其真值, 所以我们称之为命题变项 (Propositional variable).
\end{definition}

\begin{definition}{否命题}
	设有一命题 $p$, 则 $p$ 的否定 (Inverse) 记为 $\neg p$ (也可记为 $\overline p$),  $p$ 的真值与 $\neg p$ 的真值相反.
\end{definition}


\begin{definition}{合取}
设 $p, q$ 为命题,  $p, q$ 的合取式为复合命题, 记为 $p\land q$ , $\land$ 是合取联接词 (Conjunction), 即 $p, q$ 均为 T 时 $p\land q$ 取 T , 否则 $p\land q$ 取 F.
\end{definition}

\begin{definition}{析取}
设 $p, q$ 为命题,  $p, q$ 的析取式为复合命题, 记为 $p\lor q$ , $\lor$ 是析取联接词 (Disjunction), 即 $p, q$ 均为 F 时 $p\lor q$ 取 F , 否则 $p\lor q$ 取 T.
\end{definition}

\begin{definition}{异或}{eor}
设 $p, q$ 为命题, $p, q$ 的异或为复合命题 (Compound proposition), 记为 $p\oplus q$ , $\oplus$ 是异或联接词 (Exclusive OR), 即 $p, q$ 的真值不同时 $p\oplus q$ 取 T , 否则 $p\oplus q$ 取 F.
\end{definition}

\subsection{条件语句}

\begin{definition}{蕴涵}
设 $p, q$ 为命题, 命题 $p \rightarrow q$ 为命题 “若 $p$ , 则 $q$ ”, $\rightarrow$ 是蕴涵联接词 (Entailment). 当 $p$ 为真而 $q$ 为假时, 命题 $p \rightarrow q$ 为假, 否则 $p \rightarrow q$ 为真, 对于命题 $p \rightarrow q$, 其中 $p$ 为前提, $q$ 为结论.
\end{definition}

\begin{remark}
为防止后续极限符号 $\to$ 与蕴涵符号产生歧义, 在后续分析学章节中将使用 $\Rightarrow$ 符号, 仅在讨论逻辑学时使用 $\rightarrow$.	
\end{remark}

\begin{definition}{逆, 倒置与反}
设 $p, q$ 为命题, 对于条件语句 $p \rightarrow q$ , 称 $q \rightarrow p$ 为 $p \rightarrow q$ 的逆蕴涵, 称 $\neg q \rightarrow \neg p$ 为 $p \rightarrow q$ 的倒置蕴涵, 称 $\neg p \rightarrow \neg q$ 为 $p \rightarrow q$ 的反蕴涵.
\end{definition}

\begin{definition}{等价}
设 $p, q$ 为命题, 命题 $p \leftrightarrow q$ 为命题 “$p$ 当且仅当 $q$ ”, 我们也将当且仅当写作 iff (if and only if),  $\leftrightarrow$ 是等价联接词.当 $p$ 与 $q$ 的真值相等时, 命题 $p \leftrightarrow q$ 为真 , 否则命题 $p \leftrightarrow q$ 为假.
\end{definition}

\begin{remark}
在后续分析学章节中将使用 $\Leftrightarrow$ 符号表示等价, 仅在讨论逻辑学时使用 $\leftrightarrow$.	
\end{remark}

\section{命题公式与等值演算}

\subsection{命题公式与分类}

\begin{definition}{命题公式}
为了更深入研究, 将复合命题中的命题常项换为命题变项, 称这样的式子为合式公式, 合式公式由命题变项、联接词、括号组成.
\ITEMS
	\item 单个命题变项是合式公式.
	\item 如果 $A$ 是合式公式, 则 $\neg A$ 也是合式公式.
	\item 如果 $A, B$ 是合式公式, 则用联接词联接 $A, B$ 也是合式公式.
	\item 只有有限次地应用 1, 2, 3 组成的式子才是合式公式.
\ITEME
在命题逻辑中, 称合式公式为命题公式.
\end{definition}


\begin{definition}{赋值}
设 $A$ 为一个命题公式, $p_1, p_2, \cdots, p_n$ 为出现在 $A$ 中的所有命题变项.给 $p_1, p_2, \cdots, p_n$  指定一组真值, 称为对 $A$ 的一个赋值或解释.若指定的一组值使 $A$ 的值为真, 则称这组值为 $A$ 的成真赋值, 反之为成假赋值.
\end{definition}

\subsection{命题公式的分类}

\begin{definition}{命题公式的分类}
设 $A$ 为一个命题公式
\ITEMS
	\item 若 $A$ 在所有赋值下取之均为真, 则称 $A$ 为重言式或永真式.
	\item 若 $A$ 在所有赋值下取之均为假, 则称 $A$ 为矛盾式或永假式.
	\item 若 $A$ 至少存在一组成真赋值, 则称 $A$ 是可满足式.
\ITEME
\end{definition}

\subsection{等值演算}

\begin{definition}{等值演算}
设 $A, B$ 是两个命题公式, 若等价式 $A \leftrightarrow B$ 是重言式, 则称 $A, B$ 是等值的, 记作 $A\Leftrightarrow B$ .

根据已知的等值式, 推演出与给定公式等值的公式的过程称为等值演算.
\end{definition}

\begin{remark}
在讨论逻辑学时 $\leftrightarrow, \Leftrightarrow$ 两个符号的含义不同, $\Leftrightarrow$ 并不是联接词, 但在分析学时我们会使用 $\Leftrightarrow$.
\end{remark}


\begin{theorem}{}
设 $\Phi(A)$ 是含命题公式 $A$ 的命题公式, $\Phi(B)$ 是用命题公式 $B$ 置换了 $\Phi(A)$ 中的 $A$ 之后得到的命题公式.如果 $A\Leftrightarrow B$ , 则 $\Phi(A)\Leftrightarrow\Phi(B)$.
\end{theorem}

\section{范式}

\subsection{范式}

\begin{definition}{范式}
仅由有限个命题变项或其否定构成的析取式称为简单析取式, 仅由有限个命题变项或其否定构成的合取式称为简单合取式.

仅由有限个简单合取式构成的析取式称为析取范式, 仅由有限个简单析取式构成的合取式称为合取范式.
\end{definition}

\begin{theorem}{范式存在定理}{EToNF}
任意命题公式都存在与之等值的析取范式和合取范式.
\end{theorem}

\subsection{主范式}
\begin{definition}{主析取范式}
设有 $n$ 个命题变项, 若在简单合取式中每个命题变项与其否定有且仅有一个出现一次, 则这样的简单合取式称为极小项.

如果公式 $A$ 的析取范式中的简单合取式全是极小项, 则称该析取范式为 $A$ 的主析取范式.
\end{definition}

\begin{theorem}
任意命题公式都有唯一的主析取范式.
\end{theorem}

\begin{definition}{主合取范式}
设有 $n$ 个命题变项, 若在简单析取式中每个命题变项与其否定有且仅有一个出现一次, 则这样的简单析取式称为极大项.

如果公式 $A$ 的合取范式中的简单析取式全是极大项, 则称该合取范式为 $A$ 的主合取范式.
\end{definition}

\begin{theorem}
任意命题公式都有唯一的主合取范式.
\end{theorem}

\section{一阶逻辑基本概念}

\begin{definition}{个体词}
个体词是指可以独立存在的个体, 可以是具体的事物, 也可以是抽象的概念.表示具体的或特定个体的词称为个体常项, 表示抽象的或泛指的个体的词称为个体变项.个体变项的取值范围称为个体域.
\end{definition}


\begin{definition}{谓词}
谓词是用来刻画个体词的性质或个体词之间关系的词.表示具体性质和关系的谓词称为谓词常项, 表示抽象的或泛指的谓词称为谓词变项.谓词中包含的个体词数称为元数, 称包含 $n$ 个个体词的谓词为 $n$ 元谓词.
\end{definition}


\begin{definition}{量词}{exists}
量词指的是表示数量的词, 有全称量词和存在量词两种.符号 $\forall$ 表示全称量词, 读作“任意”, 符号取自英文词汇 Arbitrary 的首字母.$\forall$ 用于表达某指定范围内“所有”、“任意”或“每一个”对象都具备某种性质.符号 $\exists$ 表示存在量词, 读作“存在”, 符号取自英文词汇 Exists 的首字母.$\exists$ 用于表达某指定范围内“至少存在一个”对象具备某种性质.

特别的, 记 $\exists !$ 为唯一存在量词, 用于表达某指定范围内“唯一存在一个”对象具备某种性质.
\end{definition}

\chapter{集合论与实数理论}

集合论 (Set theory), 是研究集合(由一堆抽象对象构成的整体)的数学理论, 包含集合和元素 (或称为成员)、关系等最基本数学概念.集合论、命题逻辑与谓词逻辑共同构成了数学的公理化基础, 以未定义的“集合”与“集合成员”等术语来形式化地建构数学对象.

德国数学家 Cantor 于十九世纪七十年代提出了朴素集合论, 并从此展开了对集合论的研究.

\section{集合及其基本运算}

凡是学习过高中数学的同学应该都对朴素集合论有相关概念, 在这里我们先回顾一下中学的朴素集合论.

\subsection{朴素集合论与集合运算}

Cantor 集合论, 也被称为“朴素”集合论, 我们将其基本前提总结为以下几点

\ITEMS
	\item 互异性 \quad 集合可有任何有区别的对象组成.
	\item 无序性 \quad 集合由其组成对象整体唯一确定.
	\item 确定性 \quad 任何性质都确定一个具有该性质的对象的集合.
\ITEME


我们将组成一个集合的对象称为该集合的元素, 元素 $x$ 属于集合 $X$ 可以简记为
\[
	x\in X\ \mbox{或}\ X\ni x 
\]
其否命题为
\[
	x\notin X\ \mbox{或}\ X \not\ni x.
\]

如果集合 $A$ 的每一个元素都是 $B$ 的元素, 就说 $A$ 是 $B$ 的子集, 记作
\[
A \subseteq B.
\]
特别的，如果 $A = B$, 那么 $A \subseteq B$ 与 $B \subseteq A$ 均成立. 当 $A \subseteq B$ 并且 $A$ 不等于 $B$ 时, 称 $A$ 为 $B$ 的真子集，记作
\[
	A\subsetneq B \ \text{或者}\ A \subset B.
\]

其中 $\subseteq$ 和 $\subsetneq$ 分别称为包含关系和真包含关系.

\begin{remark}
	以上符号在使用上依地区年代有所区别，具体使用视情况而定，例如用 $\subset$ 符号表示包含，用 $\subsetneqq$ 符号表示真包含。
\end{remark}


% 简单的集合运算
对于集合 $A$ 与 $B$ 我们定义以下运算
\ITEMS
	\item 集合 $A$ 与 $B$ 的并集是集合
	\[
		A\cup B := \{x\in U\mid (x\in A) \vee (x\in B)\}.
	\]
	\item 集合 $A$ 与 $B$ 的交集是集合
	\[
		A\cap B := \{x\in U\mid (x\in A) \land (x\in B)\}.
	\]
	% 在书中补集是作为差集的延伸提出的, 但我们往往把补集视为基础运算的组合
	\item 集合 $A$ 在全集 $U$ 上的补集是集合
	\[
		A^c = C_U A := \{x\in U\mid (x\notin A) \}.
	\]
	\item 集合 $A$ 与 $B$ 的差集是集合
	\[
		A\setminus B := \{x\in U\mid (x\in A) \land (x\notin B)\}.
	\]
	\item 集合的直积(Cartesian 积)为如下集合
	\[
		A \times B :=\{(x, y)\mid (x \in A)\land(y \in B)\}.
	\]
	其中 $(x, y)$ 为一个序偶, 序偶由两个对象按照特定先后顺序排列构成的组合.
	\item 集合的对称差为如下集合 (使用了 \hyperref[def:eor]{异或运算定义})
	\[
		A \Delta B := \{x\in U\mid (x \in A) \oplus (x \in B)\}.
	\]
\ITEME

%\begin{theorem}{A. De Morgan 法则}

由集合的交并补运算可以得到 De Morgan 法则, 德摩根法则一般有两种表述, 分别为命题逻辑和集合论形式, 其中命题逻辑形式:
	\[
		\neg (p\land q)\equiv (\neg p)\lor (\neg q),
	\]
	\[
		\neg (p\lor q)\equiv (\neg p)\land (\neg q).
	\]
集合论形式:
	\[
		(A\cup B)^c = A^c\cap B^c,
	\]
	\[
		(A\cap B)^c = A^c\cup B^c.
	\]
%\end{theorem}

\subsection{Russel 悖论与数学危机}

Russel 悖论设集合 $A=\{x\mid x\in A\}$, 那么我们可以假设 $A\in A\Rightarrow A\notin A$, 同样 $A\notin A \Rightarrow A\in A$, 我们得出了截然不同的两种结果. 这意味着现代数学基础的集合论存在瑕疵, 数学家们随即对集合论进行了严格的约束, 从而规避了逻辑悖论, 但实际上数学危机远远不只是 Russel 悖论这么简单的问题, 而且仍并未被 ``完美" 解决. 

实际上的数学危机是一次将数学界分为三个学派的巨大分歧, 以 Russel 为代表的逻辑主义, 以布劳威尔为首的直觉主义, 以及 Hilbert 所率领的形式主义. 其中为主导的 Hilbert 与他的形式主义曾提出了著名的 Hilbert 纲领, 试图通过形式化的方法解决数学危机, 但之后的哥德尔不完备定理却表明任何系统都无法做到既完全又自洽, 击碎了 Hilbert 纲领的设想. 但 Hilbert 纲领深刻地重塑了现代数学与逻辑学, 我们也常称 Hilbert 为现代数学之父. 虽然数学不得已从追求绝对真理撤退到现代数学体系, 但却是我们对人类理性边界的认知的提升.

\subsection{公理化集合论}

实际上公理化集合论是一个没有数学美感的产物, 就像东拼西凑出的支柱, 主要为防止如 Russel 悖论这种矛盾的产生, 其中我们最常用的是策梅洛-弗伦克尔公理系统(Zermelo-Fraenkel), 一般简称为 ZFC 集合论.

首先我们引入类的概念, 可以将其理解为更一般化的集合, 即任何集合都是类, 但是类不一定是集合. 
\begin{definition}{类}
	如果 $\varphi(x,p_1,\cdots,p_n)$ 是表达式, 则称
	\[
		C := {x\mid\varphi(x,p_1,\cdots,p_n)}
	\]
	为类 (Class), 有 $x \in C \iff \varphi(x,p_1,\cdots,p_n)$.
\end{definition}

\begin{definition}{万有类}
	万有类 (Universal class) 是所有集合的类,
	\[
		U := \{x \mid x = x\}.
	\]
\end{definition}

\begin{definition}{包含与类运算}
	类的包含关系 (Inclusion) 如下 
	\[
		C \subseteq D \iff \forall x (x \in C \rightarrow x \in D)
	\]
	类的运算如下
	\[
		C \cap D := \{x \mid(x \in C) \land (x \in D)\},
	\]
	\[
		C \cup D := \{x \mid (x \in C) \lor (x \in D)\},
	\]
	\[
		C \setminus D := \{x \mid (x \in C) \land (x \notin D)\},
	\]
	\[
		\bigcup C := \{x \mid \exists S \in C (x \in S)\}.
	\]
\end{definition}

\begin{definition}{真类}
	称不是集合的类为真类 (Proper class). 例如 Russel 悖论所构建的就是一个真类.
\end{definition}

接下来我们来逐步构建公理化集合论, 先阐述下我们的基本思想, 首先希望让集合论支持以往的集合运算, 在此基础上尽量从根源上避免如 Russel 悖论等悖论的诞生.

\begin{axiom}{外延公理 (Axiom of extensionality)}{zfc1}
	集合 $A$ 与集合 $B$ 相等, 当且仅当它们所具有的各元素是相同的, 即
	\[
		A = B \iff A\subset B \land A\supset B.
	\]
\end{axiom}

外延公理使得集合由构成集合的元素决定.

\begin{axiom}{分离公理模式 (axiom schema of separation)}{zfc2}
	设有表达式 $\varphi (u,p_1,\cdots,p_n)$, 则对任何集合 $X$ 和任意 $p_1,\cdots ,p_n$ 存在 $Y = \{u \in X \mid \varphi(u,p_1,\cdots,p_n)\}.$
	\[
		\forall X , \forall p_{1} , \ldots , \forall p_{n} , \exists Y , \forall x \left(x \in Y \longleftrightarrow x \in X \land \varphi \left(x , p_{1} , \ldots , p_{n}\right)\right).
	\]
\end{axiom}

\begin{remark}
分离公理模式是一个公理模式, 即对于每个不同的 $\varphi$ 都有可以生成一个公理, 所以分离公理模式本质是无穷条公理.	
\end{remark}

由分离公理模式可得以下定义

% 集合运算 1
\begin{definition}{交集与差集}
	设有集合 $X, Y$ 应用分离公理模式可知, 如下集合存在,
	\[
		X \cap Y := \{u \in X \mid u \in Y\},\quad X\setminus Y := \{u \in X \mid u \notin Y\},
	\]
	分别称为集合的交集, 差集 (补集即在全集上的差集).
\end{definition}
	
但是注意到分离公理虽然为交集和补集提供了公理支持, 但是却无法支持并集运算.
	
\begin{axiom}{并集公理 (Axiom of union)}{zfc3}
	对于集合的任何集合 $M$, 存在一个集合 $M$ 的并集的集合 $\cup M$, 其元素是且仅是 $M$ 的元素所包含的那些元素.
\end{axiom}
	
由并集公理我们可以知道假如只包含集合 $X, Y$ 的集合 $\{X, Y\}$ 存在, 那么两集合的并集 $X \cup Y$ 就存在. 但注意到有需要集合 $\{X, Y\}$ 存在的限制存在, 于是提出配对公理.

\begin{axiom}{配对公理 (Axiom of pairing)}{zfc4}
	对于任何集合 $X$ 和 $Y$ , 存在一个集合 $Z$ , 其元素仅为 $X$ 和 $Y$.
\end{axiom}

\begin{definition}{对}
	由外延公理可知配对公理中的 $Z$ 是唯一的, 则对任意 $a, b$ 有对 $\{a,b\}$ 存在,
	\[
		\{a,b\} := \text{唯一的 $c$ 满足}\; \forall x(x \in c \iff (x = a) \lor (x = b)).
	\]
	单点集 $\{a\}$ 可由定义为 $\{a\} = \{a, a\}$.
\end{definition}
	
把配对公理与并集公理凑到一起我们就终于为并集运算铺平了道路, 于是有如下定义.

% 集合运算 2
\begin{definition}{并集与对称差}
	设有集合 $X, Y$ 应用配对公理与并集公理可知, 如下集合存在,
	\[
		X \cup Y := \bigcup\{X,Y\}, \quad X \Delta Y := (X \setminus Y) \cup (Y \setminus X),
	\]
	分别称为集合的并集和对称差.
\end{definition}

\begin{axiom}{空集公理}{zfc5}
	存在一个不包含任何元素的集合 $\varnothing$. 由分离定理从任意集合都可以分离出空子集, 并根据外延定理可知所有空子集都相等.
\end{axiom}

有时不将空集公理列入 ZFC 集合论, 而是将其视为分离公理模式的推论.

\begin{definition}{子集}
集合 $X$ 称为 $Y$ 的子集 (Subset), 记作 $X \subset Y$, 如果
	\[
		\forall x(x \in X \rightarrow x \in Y).
	\]
	若 $X \subseteq Y$ 且 $X \neq Y$, 称 $X$ 是 $Y$ 的真子集 (Proper subset).
\end{definition}

\begin{definition}{幂集}
	设有集合 $X$ , 则其所有子集构成的集合
	\[
		\mathcal{P}(X) := \{u\mid u \subseteq X\},
	\]
称为 $X$ 的幂集 (Power set).
\end{definition}

\begin{axiom}{幂集公理 子集之集 (Axiom of power set)}{zfc6}
	对于任何集合 $X$, 幂集 $\mathcal{P}(X)$ 存在.
\end{axiom}


在介绍无穷公理之前我们首先需要引入后继集的概念。 
\begin{definition}
	设有集合 $X$, 称 
	\[
		X^+=X\cup \{X\}
	\]
	为 $X$ 的后继集. 此外, 如果一个集合包含空集以及自身任何一个元素的后继集, 我们称该集合为归纳集. 即设有集合 $S$ 满足
	\[
		(\varnothing \in S) \land ((\forall x \in S) \ x \cup {x} \in S).
	\]
	则称 $S$ 为归纳集.
\end{definition}

\begin{axiom}{无穷公理 (Axiom of infinity)}{zfc7}
	归纳集存在.
\end{axiom}

无穷公理看起来简单, 但实际上无穷公理为现代数学的诸多研究提供了公理保障, 它使实无穷成为了一个可被研究的数学对象. 例如自然数集由此成为了合法的数学对象, 我们在无穷公理的基础上再通过之前所用的公理就可以建立自然数集 $\mathbb{N}_0$ 的模型(冯诺伊曼表示法), 对自然数作如下表示.
	\[
		\begin{aligned}
			0:\quad & \varnothing,\\
			1:\quad & \varnothing^+=\{\varnothing\},\\
			2:\quad & \{\varnothing\}^+=\{\{\varnothing\}\},\\
			\vdots & 
		\end{aligned}
	\]
	
\begin{axiom}{替换公理模式 (axiom schema of replacement)}{zfc8}
	设有表达式 $\varphi (x, y, z)$, 对于任意集合 $X$, 若对每一个 $x \in X$ 都有唯一的 $y$ 使 $\varphi (x, y, z)$ 成立, 那么就存在集合 $Y$ 恰好由所有 $y$ 构成.
\end{axiom}

实际上由替换公理模式可以推出分离公理模式, 所以有时 ZFC 不会包含分离公理模式.

\begin{axiom}{正则公理 (Axiom of regularity)}{zfc9}
	对任意非空集合 $A$, 都存在一个元素 $x \in A$, 使得 $x$ 与 $A$ 没有公共元素, 我们称该元素为 $\in$-极小元素.
	\[
		\forall A\Bigl(A\neq\varnothing\to \exists x\in A\,(x\cap A=\varnothing)\Bigr).
	\]
\end{axiom}

	
\begin{axiom}{* 选择公理 策梅洛公理 (Axiom of choice)}{zfc10}
	对于任意互不相交的非空集合组成的集合族, 存在集合 $C$, 使得对于该集合族中的任何集合 $X$, 集合 $X\cap C$ 只有一个元素组成.
\end{axiom}

ZF 集合论不包含选择公理, 因为选择公理存在一些争议, 例如承认此公理会引出分球怪论, 但由于选择公理可以为证明提供很多便利, 我们仍主要使用 ZFC 集合论.

\subsection{集合的任意交和并}

\begin{definition}{集族}
通常我们把一个以集合为元素的集合称为集族, 一般用大写花体字母表示, 如 $\mathscr{A}$.
\end{definition}

这里给出由集族概念所得的序偶定义.

\begin{definition}{序偶}
我们定义以下集族为序偶
\[
	(a,b) := \{\{a\},\{a,b\}\}.
\]
按照定义, $(a,b)$ 是两个集合的族, 分别为单元素集和两元素集, 而序偶的第一元素是同时属于这两个集合的元素, 第二个元素是仅属于一个集合的元素.
\end{definition}

对于序偶我们也可以将其作为原始概念对待. 

\begin{definition}{序偶}
序偶是指由两个元素 $x$ 和 $y$ 按一定顺序排列构成的二元组, 记作 $\langle x, y \rangle$ 或 $(x, y)$, 其中，$x$ 称为第一元素(或左投影), $y$ 称为第二元素(或右投影). 在序偶中，x 称为第一元素(或左投影), y 称为第二元素(或右投影). 序偶具有顺序性，即当 $x \neq y$ 时，$(x,y) \neq (y,x)$.
\end{definition}

我们可以使用嵌套序偶构造多元组, 例如 $(x,y,z) = ((x,y),z)$ 为三元组, 显然元组也是有序的.

\begin{definition}{加标集族}
设 $J$ 是一个非空集合, 若对于每个 $\alpha \in J$ 都有集合 $A_\alpha$. 则由所有 $A_\alpha$ 构成的集族 $\mathscr{A}$ 称为加标集族, 定义如下
\[
	\mathscr{A} = \{A_\alpha\}_{\alpha \in J} := \{A_\alpha\mid \alpha \in J \}.
\]
读作 ``$\alpha$ 取遍 $J$ 时, 所有 $A_\alpha$ 的集族". 若指标集自明, 则也可以记为 $\{A_\alpha\}$.
\end{definition}

\begin{definition}{Cartesian 乘积}
	设有集合 $A, B$, 则其的 Cartesian 乘积 (Cartesian product) 为
	\[
		A \times B := \{(a,b)\mid(a \in A) \lor (b \in B)\}.
	\]
\end{definition}


\begin{definition}{集族中的交和并}
	令 $X$ 是一集合且 $\mathscr{A} = \{A_\alpha \mid\alpha \in J\}$ 是由 $X$ 的一些子集所构成的集族. 定义集合族 $\mathscr{A}$ 中元素的并为
	\[
		\bigcup_{\alpha \in J} A_\alpha \equiv \bigcup_{A \in \mathscr{A}} A := \{x \mid \exists A \in \mathscr{A} (x \in A)\},
	\]
	定义集合族 $\mathscr{A}$ 中元素的交为
	\[
		\bigcap_{\alpha \in J} A_\alpha \equiv \bigcap_{A \in \mathscr{A}} A := \{x \mid \forall A \in \mathscr{A} (x \in A)\}.
	\]
\end{definition}

\begin{theorem}{运算律}
	若 $\mathscr{A} = \{A_\alpha \mid \alpha \in J_A\}$ 和 $\mathscr{B} = \{B_\beta \mid \beta \in J_B\}$ 是由集合 $X$ 的一些子集所构成的集族, 则
	\ITEMS
		\item 结合率
		\[
			\bigcap_{\alpha\in J_A} A_\alpha \cap \bigcap_{\beta\in J_B} B_\beta = \bigcap_{(\alpha,\beta)\in J_A \times J_B} A_\alpha \cap B_\beta,
		\]
		\[
			\bigcup_{\alpha\in J_A} A_\alpha \cup \bigcup_{\beta\in J_B} B_\beta = \bigcup_{(\alpha,\beta)\in J_A \times J_B} A_\alpha \cup B_\beta.
		\]
		\item 分配律
		\[
			\bigcap_{\alpha\in J_A} A_\alpha \cup \bigcap_{\beta\in J_B} B_\beta = \bigcap_{(\alpha,\beta)\in J_A \times J_B} A_\alpha \cup B_\beta,
		\]
		\[
			\bigcup_{\alpha\in J_A} A_\alpha \cap \bigcup_{\beta\in J_B} B_\beta = \bigcup_{(\alpha,\beta)\in J_A \times J_B} A_\alpha \cap B_\beta.
		\]
		\item De Morgan 法则
		\[
			\left(\bigcup_{\alpha\in J_A} A_\alpha\right)^c = \bigcap_{\alpha\in J_A} A_\alpha^c,
		\]
		\[
			\left(\bigcap_{\alpha\in J_A} A_\alpha\right)^c = \bigcup_{\alpha\in J_A} A_\alpha^c.
		\]
	\ITEME
\end{theorem}

\section{映射与关系}

在高中我们就学习过函数的定义, 不过实际上函数的严格定义诞生的时间在数学史上很晚, Leibniz 在 1673 年才首次使用了术语 ``函数", 这时还没有对函数的严格数学定义, 但随着用代数方法研究曲线的发展, 用一个术语来表示“利用解析表达式来表示依赖于一个变量的数量”变得越来越有必要了. 在 Leibniz 和 Jean Bernoulli 的 1695 年和 1698 年的通信中, ``函数" 最终被采纳. 1718 年, Jean Bernoulli 将变量的函数定义为以某种方式由变量和常量组成的量.

\subsection{映射}

映射是更抽象化的函数, 将函数从数集拓展到一般集合上.

\begin{definition}{赋值法则}
	设有集合 $C, D$, 一个赋值法则 (Rule of assignment) $R$ 是指满足条件
	\[
		\bigl((c, d) \in R \bigr) \land \bigl((c, d') \in R \bigr)\Rightarrow d = d'
	\]
	的 $C \times D$ 的子集. 赋值法则的定义域 (Domain) 和像域 (Image set) 如下
	\[
		D_R := \{c\in C\mid\exists d\in D \bigl((c,d) \in R\bigr)\}, 
	\]
	\[
		I_R := \{d\in D\mid\exists c\in C \bigl((c,d) \in R\bigr)\}.
	\]
\end{definition}

\begin{definition}{映射}

	对于一个赋值法则 $R$, 与集合 $A, B$ 若
	\[
		R \subseteq D_R \times I_R \subseteq A \times B,
	\]
	则我们称三元组 $(R, A, B)$ 为一个映射 (mapping) $f$, 其中 $B$ 称为陪域, $A$ 为定义域, 当 $A$ 由 $R$ 可得是也可以用序偶 $(R, B)$ 表示. 通常用以下符号表示映射, (此符号由 Eular 引入)
	\[
	\begin{aligned}
		f:X&\rightarrow Y,\\
		x&\mapsto f(x).
	\end{aligned}
	\]
	定义映射的定义域与值域如下
	\[
		D_f := D_R,\qquad I_f := I_R.
	\]
	$f$ 的图 (graph) 定义为
	\[
		\operatorname{graph}(f) := \{(a,f(a))\mid a \in A\} \subseteq A \times B.
	\]
\end{definition}

\begin{definition}{限制与延拓}
设有集合 $X, Y$, 有映射 $f:X\to Y$, 对于 $A \subseteq X$ 构建如下映射
\[
	f|_A : X\to Y.
\]
映射 $f|_A$ 与 $f$ 在集合 $A$ 下相等, 即在每个元素 $x\in A$ 上映射所得的像相等.我们称 $f|_A$ 为 $f$ 在 $A$ 上的限制, 而相对于 $f|_A$ 称 $f$ 为 $f|_A$ 在 $X$ 上的延拓.
\end{definition}

\begin{definition}{映射的分类}
满足不同元素具有不同的像的映射称为单射 (Injective), 即
\[
	x_1 \ne x_2 \Rightarrow f(x_1) \ne f(x_2).
\]

满足映射的像等于集合 $Y$ 的映射称为满射 (Surjective), 即
\[
	f(X)=Y.
\]

即是单射又是满射的映射的映射是双射 (Bijective) , 即
\[
\begin{aligned}
	&f(X)=Y,\\
	&(f(x_1) = f(x_2)) \Rightarrow(x_1=x_2).
\end{aligned}
\]
\end{definition}

\begin{definition}{逆映射}
当映射是双射的时候, 存在 $f:A\rightarrow B$ 的逆映射 (Inverse), 记为
\[
	f^{-1}: B \rightarrow A.
\]
给定映射 $f : A \rightarrow B$. 定义 $f$ 的左逆 (Left inverse) 存在 $g : B \rightarrow A$ 对所有 $a \in A$ 都有
\[
	g(f(a)) = a.
\] 
定义 $f$ 的右逆 (Right inverse) 存在 $h: B \rightarrow A$ 对所有 $b\in B$ 都有 
\[
	f(h(b)) = b.
\]
\end{definition}

\begin{lemma}
	给定映射 $f : A \rightarrow B$. 如果存在 $f$ 的左逆 $g : B \rightarrow A$ 和 $f$ 的右逆 $h: B \rightarrow A$, 则 $g = h = f^{−1}$.
\end{lemma}

\begin{definition}{映射的复合运算}
如果在映射 $f:X\to Y$ 和 $g:Y\to Z$ 中, 构建一个新的映射
\[
\begin{aligned}
	g\circ f:X&\to Z\\
	x &\mapsto g(f(x)).
\end{aligned}
\]
这样构建出的映射 $g\circ f$ 称为映射 $f$ 与 $g$ 的复合映射.
\end{definition}

\subsection{关系}

关系是为了将 ``=" 和 ``<" 从定义在数集上推广到定义在一般集合上.

\begin{definition}{二元关系}
关系 (relation) 是两个集合 Cartesian 积的子集.如果有两个集合 $A$ 和 $B$, 它们之间的二元关系 $\mathcal{R}$ 是集合 $A$ 与 $B$ 的 Cartesian 积的子集, 即 $\mathcal{R} \subseteq A \times B$.若元素 $a \in A$ 与 $b \in B$ 满足 $(a,  b) \in \mathcal{R}$, 则记作 $a\mathcal{R}b$, 否则记为 $a\overline{\mathcal{R}}b$.对于集合 $X$, 如果 $\mathcal{R} \subseteq X^2$, 就说在 $X$ 上给定了二元关系 $\mathcal{R}$.
\end{definition}

\begin{definition}{等价关系}
对于一非空集合 $X$, 若在 $X$ 上给定了二元关系 $\mathcal{R}$ 满足以下条件
\ITEMS
	\item 自反性 (Reflexivity)\quad $\forall x\in X \Rightarrow x\mathcal{R}x$,
	\item 对称性 (Symmetry)\quad 若 $x, y\in X ,  x\mathcal{R}y$ 则 $y\mathcal{R}x$,
	\item 传递性 (Transitivity)\quad 若 $x, y, z\in X ,  x\mathcal{R}y,  y\mathcal{R}z$ 则 $x\mathcal{R}z$.
\ITEME 
则称该二元关系为等价关系 (Equivalence relation), 记作 $x\sim y$.
\end{definition}

\begin{definition}{分类}
若把集合 $M$ 的全体元素分成若干互不相交的子集, 这些子集的并集等于 $A$, 则由这些子集构成的集族叫做 $M$ 的一个分类 (partition).
\end{definition}

\begin{theorem}
	集合 $M$ 的一个分类决定 $M$ 的一个等价关系.
\end{theorem}
\begin{proof}
	对集合 $M$ 的元素规定以下关系
	\[
		a\mathcal{R}b \iff a \text{ 与\ } b \text{ 同在一类}.
	\]
	这显然是 $M$ 的一个等价关系.
\end{proof}
\smallskip

\begin{theorem}
	集合 $M$ 的一个等价关系决定 $M$ 的一个分类.
\end{theorem}
\begin{proof}
	下面利用 $M$ 的等价关系 $\sim$ 来作出 $M$ 的一个分类.
	
	任取 $a\in M$, 与 $a$ 等价的一切元素作成 $M$ 的一个子集, 这个子集记为 $\overline{a}$.由于 $a\sim a$, 故 $a \in \overline{a}$, 因此, $M$ 中每个元素都一定属于一个类.
	
	下面再证每个元素只属于一个类.
	
	设 $a\in\overline{b}, a\in\overline{c}$, 则
	\[
		a\sim b ,  \ \ a\sim c.
	\]
	任取 $x \in \overline{b}$, 则 $x\sim b$.由上可得 
	\[
		b\sim a ,  \ \ a\sim c.
	\]
	从而 $x\sim c$, 即 $x\in\overline{c}$, 于是 $\overline{b} \subseteq\overline{c}$.
	
	同理可证 $\overline{c} \subseteq\overline{b}$, 因此 $\overline{b}=\overline{c}$.即如果两个类公有元素 $a$, 则两个类完全相等, 说明每个元素只能属于一个类. 
	
\end{proof}
\smallskip

\begin{definition}{等价类}
	对于给定集合 $X$ 中的一个等价关系, 及一元素 $\forall x\in X$, 我们称由 $x$ 决定的等价类为
	\[
		[x] := \{y\in X:y\sim x\}.
	\]
	其中由 $x$ 决定的等价类包含 $x$.
\end{definition}

\begin{definition}{序关系}
对于一非空集合 $X$, 若在 $X$ 上给定了二元关系 $\mathcal{R}$ 满足以下条件
\ITEMS
	\item 相容性 (Comparability)\quad $ \forall x, y \in A$ 且 $x \neq y$ 则 $x\mathcal{R}y$ 或 $y\mathcal{R}x$ 其一成立, 
	\item 非自反性 (Nonreflexivity)\quad $\neg\exists x\in X \Rightarrow x\mathcal{R}x$,
	\item 传递性 (Transitivity)\quad 若 $x, y, z\in X ,  x\mathcal{R}y,  y\mathcal{R}z$ 则 $x\mathcal{R}z$.
\ITEME 
则称该二元关系为序关系 (Order relation) 或简单序 (Simple order) 或线性序 (Linear order), 记为 $x < y$. 特别的 $x \le y$ 表示 $x < y$ 或 $x = y$. 若 $X$ 上存在一个序关系 $<$, 则 $(X,<)$ 是有序集 (Ordered set).
\end{definition}

% 严格偏序关系
\begin{definition}{严格偏序关系}
对于一非空集合 $X$, 若在 $X$ 上给定了二元关系 $\mathcal{R}$ 满足以下条件
\ITEMS
	\item 非自反性\quad \quad $\neg\exists x\in X \Rightarrow x\mathcal{R}x$,
	\item 传递性\quad 若 $x, y, z\in X ,  x\mathcal{R}y,  y\mathcal{R}z$ 则 $x\mathcal{R}z$.
\ITEME 
则称该二元关系为严格偏序关系 (Strict partial order) , 记作 $x \prec y$, 显然序关系必是严格偏序关系. 

类似的如果 $\prec$ 是 $A$ 上的严格偏序关系, 则 $(A,\prec)$ 为严格偏序集 (Strictly partially ordered set). 对于严格偏序集 $(A,\prec)$, 设 $A_0$ 是 $A$ 的子集, 若不存在 $x \in A_0$ 满足 $b\prec x$ 则称 $b$ 是 $A_0$ 的极大元 (Maximal element). 若 $x \in A_0$ 满足 $x\prec a$. 称 $a$ 是 $A_0$ 的极小元 (Minimal element).
\end{definition}

\begin{lemma}{Zorn 引理}
假设集合 $A$ 上有一个严格偏序关系 $\prec$. 如果 $A$ 的任何有序子集在 $A$ 中都有上界, 则 $A$ 必有极大元.
\end{lemma}

% 偏序关系
\begin{definition}{偏序关系}
对于一非空集合 $X$, 若在 $X$ 上给定了二元关系 $\mathcal{R}$ 满足以下条件
\ITEMS
	\item 自反性\quad $\forall x\in X \Rightarrow x\mathcal{R}x$,
	\item 反对称性\quad 若 $x, y\in X ,  x\mathcal{R}y$ 且 $y\mathcal{R}x$ 则 $x=y$,
	\item 传递性\quad 若 $x, y, z\in X ,  x\mathcal{R}y,  y\mathcal{R}z$ 则 $x\mathcal{R}z$.
\ITEME 
则称该二元关系为偏序关系 (Partial order), 记作 $x \preccurlyeq y$. 类似的如果 $\preccurlyeq$ 是 $A$ 上的严格偏序关系, 则 $(A,\preccurlyeq)$ 为偏序集 (Partially ordered set). 假设 $\preccurlyeq$ 是 $A$ 上的偏序. 如果进一步有
\[
	\forall a,b \in A \Rightarrow (a \preccurlyeq b) \lor (b \preccurlyeq a),
\]
则称 $\preccurlyeq$ 是 $A$ 上的全序 (total order), 并称 $(A,\preccurlyeq)$ 为全序集 (totally order set). 类似地可以定义全序子集(totally order subset) 概念, 如果对任何 $a,b \in A_0$ 都有 $a \preccurlyeq b$ 或 $b \preccurlyeq a$ 成立, 则子集 $A_0 \subseteq A$ 称为全序子集.
\end{definition}



\begin{definition}{函数关系}
满足
\[
	(x \mathcal{R} y_1) \land (x \mathcal{R} y_2) \Rightarrow (y_1 = y_2).
\]
的关系 $\mathcal{R}$ 称为函数关系.
\end{definition}

我们曾在之前介绍过 Cartesian 积, 在这里我们将介绍 Cartesian 积的另一种定义.

\begin{definition}{元组}
设 $m$ 是一个正整数, 对于给定的集合 $X$, $X$ 中元素的一个 $m$ 元组定义为函数
\[
	\boldsymbol{x}:\{1,\cdots,m\}\to X.
\]
我们把 $\boldsymbol{x}$ 在 $i$ 项的值记为 $x_i$, 并称之为 $\boldsymbol{x}$ 的第 $i$ 个坐标, 函数 $\boldsymbol{x}$ 本身写为
\[
	(x_1,\cdots,x_m).
\] 
\end{definition}

\begin{definition}{序列}
给定一个集合 $X$, 定义 $X$ 中的元素的 $\omega$ 元组为函数
\[
	\boldsymbol{x}:\Z_+ \to X;
\]
我们也称 $\omega$ 元组为 $X$ 中元素的一个序列或一个无穷序列. 记 $\boldsymbol{x}$ 在 $i$ 项的值为 $x_i$, 并称之为 $\boldsymbol{x}$ 的第 $i$ 个坐标, 函数 $\boldsymbol{x}$ 本身写为
\[
	(x_1, x_2,\cdots) \text{  或者  } (x_n)_{n\in \Z_+}.
\]
\end{definition}

\begin{definition}{Cartesian 积}
设 $\{A_1,\cdots,A_m\}$ 是一个以 $\{1,\cdots,m\}$ 为指标集的加标集族. 令 $X = A_1 \cup \cdots \cup A_m$. 这个加标族的 Cartesian 积记作
\[
	\prod_{i=1}^m A_i \text{  或者  } A_1 \times \cdots \times A_m,
\]
定义为 $X$ 中元素的所有 $m$ 元组 $(x_1,\cdots,x_m)$ 的集合, 使得对每一个 $i$ 有 $x_i\in A_i$. 

\smallskip
设 $\{A_1,A_2,\cdots\}$ 是一个以正整数集作为指标集的加标集族. $X$ 为这个集族中所有集合之并. 这个加标族的 Cartesian 积记作
\[
	\prod_{i\in \Z_+} A_i \text{  或者  } A_1 \times A_2 \times \cdots,
\]
定义为 $X$ 中元素的所有 $\omega$ 元组的集合, 使得对于每一个 $i,\ x_i\in A_i$.
\end{definition}

\section{抽象代数初步}

抽象代数是最能体现现代数学思想的分支之一, 实际上抽象代数也叫近世代数. 在抽象代数中我们将脱离对具体集合与具体运算的研究, 转而研究符合指定 ``对称性"的抽象代数结构, 从具体数系和具体运算中抽象出共同结构，再用公理化的方法统一研究.

19 世纪 20 年代, Abel 严格证明了一般五次以上方程不可能用根式求解，并引入了交换群 (Abel群) 的概念. 紧随其后, 法国天才 Galois 在 1830 年前后彻底解决了方程的可解性问题: 他不仅使用了 ``群" 这一术语, 还引入了正规子群等概念, 并证明一个方程有根式解当且仅当其对应的伽罗瓦群是可解群. 伽罗瓦的理论首次揭示出, 方程求解的实质是一个结构性质问题, 从而开辟了以群论研究代数方程的全新领域.

\subsection{代数运算}

\begin{definition}{代数运算}
	设 $A, B, D$ 为集合, 称从 $A \times B$ 到 $D$ 的映射
	\[
		\circ: A \times B \rightarrow D
	\]
	为代数运算 (Algebraic operation). 特别地, 当映射为
	\[
		\circ: A \times A \rightarrow A
	\]
	时, 即为集合 $A$ 上的二元代数运算 (Binary operation).
\end{definition}

\begin{definition}{分配律}{}

\subsection{同态与同构}

\begin{definition}{同态}{}
设集合 $M$ 与 $\overline{M}$ 各有代数运算 $\circ$ 及 $\overline{\circ}$, 且 $\varphi$ 是 $M$ 到 $\overline{M}$ 的一个映射. 如果 $\varphi$ 保持运算, 即对 $M$ 中任意元素 $a, b$, 在 $\varphi$ 之下有
\[
	\varphi(a) = \overline{a}, \ \varphi(b) = \overline{b},
\]
总可得
\[
	\varphi(a \circ b) = \overline{a} \overline{\circ} \overline{b}.
\]
则称 $\varphi$ 为代数系统 $M$ 到 $\overline{M}$ 的一个同态映射 (Homomorphic mapping), 若 $\varphi$ 又是满射, 则 $\varphi$ 得一个同态满射.

如果 $M$ 到 $\overline{M}$ 存在同态满射, 则简称 $M$ 与 $\overline{M}$ 同态 (Homomorphism), 记为
\[
	M\sim\overline{M}.
\]

同态指的是一个映射保持代数结构中的运算关系, 即映射与运算的顺序可以交换.
\end{definition}

% 结合律和交换律在同态满射上的传递性
\begin{theorem}
	设集合 $M$ 与 $\overline{M}$ 分别有代数运算 $\circ$ 与 $\overline{\circ}$, 且 $M \sim \overline{M}$, 则
	\ITEMS
		\item 当 $\circ$ 满足结合律时, $\overline{\circ}$ 也满足结合律
		\item 当 $\circ$ 满足交换律时, $\overline{\circ}$ 也满足交换律
	\ITEME
\end{theorem}

\begin{proof}
	假设有同态映射 $\varphi$, 则由其定义易证.
\end{proof}


\begin{theorem}
	设集合 $M$ 有代数运算 $\circ$ 即 $\oplus$ , 集合 $\overline{M}$ 有代数运算 $\overline{\circ}$ 及 $\overline{\oplus}$；又 $\varphi$ 是 $M$ 到 $\overline{M}$ 的一个满射, 且是关于 $\circ$ 与 $\overline{\circ}$ 以及 $\oplus$ 与 $\overline{\oplus}$ 的同态, 则当 $\circ$ 对 $\oplus$ 满足左(右)分配律时, $\overline{\circ}$ 对 $\overline{\oplus}$ 也满足左(右)分配律.
\end{theorem}

\begin{proof}
	假设有同态映射 $\varphi$, 则由其定义易证.
\end{proof}

\begin{definition}{同构}
设 $\varphi$ 是 $M$ 到 $\overline{M}$ (关于代数运算 $\circ$ 及 $\overline{\circ}$) 的一个同态满射, 如果 $\varphi$ 又是单射 (即 $\varphi$ 是双射), 则称 $\varphi$ 是 $M$ 与 $\overline{M}$ 的一个同构映射 (Isomorphic mapping).

如果集合 $M$ 与 $\overline{M}$ 存在同构映射, 就说 $M$ 与 $\overline{M}$ 同构 (Isomorphism), 记为 $M \cong \overline{M}$. 否则, 即若 $M$ 到 $\overline{M}$ 不存在任何同构映射, 则称 $M$ 与 $\overline{M}$ 不同构.

$M$ 到自身的同态映射, 称为 $M$ 的自同态映射, 简称 $M$ 的自同态 (endomorphism), 同样 $M$ 到自身的同构映射, 叫做 $M$ 的自同构映射, 简称 $M$ 的自同构 (Automorphism).

同构指的是两个代数结构在结构上完全相同，只是元素的名字可能不同.
\end{definition}

同构是抽象代数中最重要的概念之一, 从运算的性质上来看两个同构的代数系统是完全相同的, 这也正好揭示了抽象代数所研究问题的实质.
\bigskip

\subsection{群}

群论始于Abel关于五次多项式方程没有一般代数解, 和Galois关于五次以上多项式方程没有一般代数解的Galois 理论(Galois theory).

\begin{axiom}{群}{group}
	设有序偶 $(G,\circ)$, 其中 $G$ 是非空集合, $\circ$ 是二元运算, 有以下公理
	\ITEMS
		\item[{\bf G$_1$}] $\circ$ 是结合的,
		\item[{\bf G$_2$}] $G$ 中有单位元 $e$ 称为 $G$ 的单位元 (Identity), 对 $\forall a \in G$ 都有
		\[
			e \circ a = a = a \circ e,
		\]
		\item[{\bf G$_3$}]  对任何 $a \in G$ 都有 $a^{-1} \in G$ 称为 $a$ 的逆元 (Inverse), 有
		\[
			a^{-1} \circ a = e = a \circ a^{-1}.
		\] 
	\ITEME 
	若 $(G,\circ)$ 满足上述三条公理, 则称其为一个群. 如果 $G$ 中的元素个数有限, 则称 $G$ 是有限的 (Finite) 并把元素个数 $\lvert G \rvert$ 称为群 $G$ 的阶 (Order). 若 $G$ 中的元素个数无限, 则称 $G$ 是无限的 (Infinite), $\lvert G \rvert$ 为无限.
\end{axiom}

\begin{remark}
	上面给出的是公理化定义, 实际上只要满足左单位元和左逆元
	\[
		e \circ a = a,\qquad a^{-1} \circ a = e,
	\]
	即可从结合律推出 $(G,\circ)$ 是群.
\end{remark}

\begin{definition}{Abelian 群}{abeliangroup}
	若有群 $(G,\circ)$ 其中 $\circ$ 是可交换的, 即对任意 $a,b \in D$ 有
	\[
		a \circ b = b \circ a,
	\]
	则称 $G$ 为 Abelian 群或交换群
\end{definition}

\subsection{环}

\begin{axiom}{环}{ring}
	设有三元组 $(R,\oplus,\otimes)$ 其中 $R$ 是一个非空集合, 其上定义了两个二元运算, 分别为 $\oplus$ (加法) 和 $\otimes$ (乘法), 其中 $a \otimes b$ 可以记为 $ab$. 有以下公理
	\ITEMS
		\item[{\bf R$_1$}] $(R,\oplus)$ 是 Abelian 群,
		\item[{\bf R$_2$}] 乘法运算 $\otimes$ 是结合的,
		\item[{\bf R$_3$}] 乘法运算 $\otimes$ 满足分配律, 即对 $\forall a,b,c \in R$ 有
		\[
			(a \oplus b) \otimes c = ac \oplus bc, \quad c \otimes(a\oplus b) = ca \oplus cb.
		\]
	\ITEME
	若 $(R,\oplus,\otimes)$ 满足上述三条公理, 则称其为一个环 (Ring). 
\end{axiom}

\begin{definition}{交换环}{commutativering}
	如果环 $(R,+,\cdot)$ 中乘法运算 $\otimes$ 是可交换的, 即对 $a,b \in \R$ 有
	\[
		a \otimes b = b \otimes a.
	\]
	则称之为交换环 (commutative ring).
\end{definition}

\subsection{域}

\begin{axiom}{域}{field}
	设有三元组 $(F,\oplus,\otimes)$ 其中 $F$ 是一个非空集合, 其上定义了两个二元运算, 分别为 $\oplus$ (加法) 和 $\otimes$ (乘法), 其中 $a \otimes b$ 可以记为 $ab$. 有以下公理
	\ITEMS
		\item[{\bf F$_1$}] $(F,\oplus)$ 是 Abelian 群, 单位元为 $0_F$,
		\item[{\bf F$_2$}] $(F\setminus\{0_F\},\otimes)$ 是 Abelian 群,
		\item[{\bf F$_3$}] 乘法运算 $\otimes$ 满足分配律, 即对 $\forall a,b,c \in R$ 有
		\[
			(a \oplus b) \otimes c = ac \oplus bc, \quad c \otimes(a\oplus b) = ca \oplus cb.
		\]
	\ITEME
	若 $(F,\oplus,\otimes)$ 满足上述三条公理, 则称其为一个域 (Field).

\end{axiom}

\section{数集公理系统与一般性质}

我们在这一节将沿着历史上数学家们对数集的探索, 来逐步学习现代的数集公理系统.

\subsection{自然数}

首先有最符合我们自然使用的自然数集, 结绳计数就是自然数集的例子.

\begin{axiom}{Dedekind–Peano 自然数公理化}{N}
自然数集 (Natural numbers) 是三元组 $(\N,0,\nu)$, 其中 $\N$ 是集合, $0 \in \N$ 是固定点, 而 $\nu$ 是如下映射
\[
	\nu: \N \rightarrow \N_*
\]
其中 $\N_* = \N\setminus\{0\}$ , 满足有以下公理
\ITEMS
	\item[{\bf N$_0$}] $\nu$ 是单射,
	\item[{\bf N$_1$}] 如果 $N \subset \N$ 是包含 $0$ 的子集且 $\nu (n) \in N$ 对所有 $n \in N$ 都成立, 则 $N = \N$.
\ITEME
\end{axiom}

上述对自然数的刻画称为 Dedekind–Peano 自然数公理化, 其等价于我们熟悉的 Peano 算数公理.

\begin{axiom}{Peano 算数公理}
如果以下条件成立, 则集合 $\N$ 称为自然数集, 其元素称为自然数.
\ITEMS
	\item $0$ 是自然数.
	\item 对于集合 $\N$ 中的任意元素 $n$, 其后继 $n'$ 也是 $\N$ 中的一个明确且唯一的自然数.
	\item $0$ 不是任何自然数的后继数.
	\item 对于每个自然数 $b, c$, 有 $b=c$ 当且仅当 $b' = c'$.
	\item 任意关于自然数的命题, 如果证明: 它对自然数 $0$ 是真的, 且假定它对自然数 $a$ 为真时, 可以证明对 $a'$ 也真, 那么命题对所有自然数都真.
\ITEME
\end{axiom}

\begin{remark}
	\hyperref[ax:N]{公理 N$_2$} (Peano 算数公理的公理 5) 保证了数学归纳原理的正确性, 是数学证明的基本工具.自然数集在某些定义下不包含 $0$, 在这里我们记不包含零的自然数集为 $\N^+$.
\end{remark}

\begin{example}
	用数学归纳原理证明牛顿二项式公式,
	\[
		(a + b)^{n} = \sum_{k = 0}^{n} {n \choose k} a^{n - k} b^{k},
	\]
	其中 ${n \choose k}$ 称为二项式系数, 其等于 $\frac {n!}{k!(n-k)!}$. 符号 $!$ 表示阶乘, 即所有小于等于该数的正整数的积.
\end{example}
\begin{solution}
	假设对于某一 $m \in \N^+$, 有
	\[
		(a + b)^{m} = \sum_{k = 0}^{m} {m \choose k} a^{m - k} b^{k}.
	\]
	
	我们要证明其对 $(a + b)^{m+1}$ 成立,
	\[
	\begin{aligned}
		(a + b)^{m+1} & = (a + b)(a + b)^{m}\\
		& = (a + b)\sum_{k = 0}^{m} {m \choose k} a^{m - k} b^{k}\\
		& = a\sum_{k = 0}^{m} {m \choose k} a^{m - k} b^{k} + b\sum_{k = 0}^{m} {m \choose k} a^{m - k} b^{k}\\
		& = \sum_{k = 0}^{m} {m \choose k} a^{m - k + 1} b^{k} + \sum_{k = 0}^{m} {m \choose k} a^{m - k} b^{k + 1} 
	\end{aligned}
	\]
	
	对于上式中 $\sum_{k = 0}^{m} {m \choose k} a^{m - k} b^{k + 1}$, 令 $k = j - 1$
	\[
		\sum_{j = 1}^{m+1} {m \choose j-1} a^{m - j + 1} b^{j}.
	\]
	
	将结果带回原式可得
	\[
		(a + b)^{m+1} = \sum_{j = 1}^{m+1} {m + 1 \choose j} a^{m - j + 1} b^{j}.
	\]
	
	因为 $(a+b)^1 = {1 \choose 1}a^1 b^0 + {1 \choose 1}a^0 b^1$ 显然成立, 根据归纳原理, 上式对所有正整数 $n$ 成立.
\end{solution}

\begin{definition}{无理数集}
不是有理数的实数称为无理数, 我们一般用符号 $\R\setminus\Q$ 表示无理数.
\end{definition}
\smallskip

在这里证明一个有意思的例子, 实数 $\sqrt 2$ 存在, 且是一个无理数.
\begin{proof}

以上命题根据无理数的定义可以写为:  $s^2 = 2,  s\in \R,  s\notin \Q$.

设 $X$ 和 $Y$ 是满足以下条件的正实数的集合:  $\forall x \in X (x^2 < 2), \ \forall y \in Y (2 < y^2)$.因为 $1 \in X,  2 \in Y$, 所以 $X$ 和 $Y$ 都不是空集.

此外, 因为对于正实数 $x$ 和 $y$ 有 $(x < y) \Leftrightarrow (x^2 < y^2)$, 所以任何元素 $x \in X$ 小于任何元素 $y \in Y$.根据完备性公理, 存在一个数 $s \in \R$, 使得对于 $\forall x \in X,  \forall y \in Y$ 有 $x \le s \le y$.

我们先证明 $s^2 = 2$, 假如 $s^2 < 2$, 因为 $1 \in X$ 那么可知 $1^2 \le s^2 < 2$, 即 $0 < \Delta = 2 - s^2 \le 1$.这意味着
\[
	\left( s + \frac{\Delta}{3s} \right)^2 = s^2 + 2\frac{\Delta}{3} + \left( \frac{\Delta}{3s} \right)^2 > s^2 + \Delta = 2.
\]
这与 $x \le s$ 相矛盾, 所以不成立.

假如 $s^2 > 2$, 因为 $2 \in Y$ 那么可知 $2 < s^2 le 2^2$, 即 $0 < \Delta = s^2 - 2 <3$.这意味着
\[
	\left( s - \frac{\Delta}{3s} \right)^2 = s^2 - 2\frac{\Delta}{3} + \left( \frac{\Delta}{3s} \right)^2 > s^2 - \Delta = 2.
\]
这与集合 $Y$ 有下界 $s$ 相矛盾, 所以不成立.

于是我们证明了 $s^2 = 2$, 接下来只需要证明 $s \notin \Q$ 即可, 假如 $s \in \Q$, 设 $m/n$ 是 $s$ 的既约分数, 则 $m^2 = 2n^2$, 所以 $m^2$ 能被 $2$ 整除, 而这也表示 $m$ 也能被 $2$ 整除.但是, 如果 $m = 2k$, 则 $2k^2 = n^2$, 而根据同样的理由 $n$ 也应当能被 $2$ 整除, 这与 $m/n$ 是 $s$ 的既约分数相矛盾.

\end{proof}

\begin{axiom}{加法公理}
	定义如下映射(加法运算)
	\[
		+:\R\times \R\rightarrow\R.
	\]
	
	1$_+$. 中性元素 $0$ 存在;, 并且对任何的 $x\in \R$, 
	\[
		x+0 = 0+x = x.
	\]
	
	2$_+$. 对于任何元素 $x\in\R$, 元素 $-x\in\R$ 一定存在, 称为 $x$ 的相反元素, 满足
	\[
		x+(-x)=(-x)+x=0.
	\]
	
	3$_+$. 运算满足结合律, 即对 $\R$ 的任何元素 $x, y, z$ 满足
	\[
		x+(y+z) = (x+y)+z.
	\]
	
	4$_+$. 运算满足交换, 即对 $\R$ 的任何元素 $x, y$ 满足
	\[
		x + y = y + x.
	\]
\end{axiom}
	
\begin{axiom}{乘法公理}
	定义如下映射(乘法运算)
	\[
		\bullet:\R\times \R\rightarrow\R.
	\]
	
	1$_\bullet$. 中性元素 $1\in\R\setminus0$ 存在, 并且对任何的 $x\in \R$, 
	\[
		x\cdot 1 = 1\cdot x = x.
	\]
	
	2$_\bullet$. 对于任何元素 $x\in\R\setminus0$, 元素 $-x\in\R$ 一定存在, 称为 $x$ 的逆元素, 满足
	\[
		x+(-x)=(-x)+x=0.
	\]
	
	3$_\bullet$. 运算满足结合律, 即对 $\R$ 的任何元素 $x, y, z$ 满足
	\[
		x\cdot(y\cdot z) = (x\cdot y)\cdot z.
	\]
	
	4$_\bullet$. 运算满足交换, 即对 $\R$ 的任何元素 $x, y$ 满足
	\[
		x \cdot y = y \cdot x.
	\]
\end{axiom}
	
\begin{axiom}[序公理]
	对于$\R$的元素$x$, $y$, 可以存在$x\le y$是否成立, 这时以下条件成立
	
	0$_\le$. $\forall x\in\R (x\le x)$.
	
	1$_\le$. $(x\le y)\land(y\le x)\Rightarrow(x=y)$.
	
	2$_\le$. $(x\le y)\land(y\le z)\Rightarrow(x\le z)$.
	
	3$_\le$. $\forall x\in\R\forall y\in\R(x\le y)\lor(y\le x)$.

\end{axiom}
	
\begin{axiom}{完备性公理}
	如果 $X$ 和 $Y$ 是 $\R$ 的非空子集, 并对于任何 $x\in X$, $y\in Y$ 有 $x \le y$, 则存在 $c\in\R$, 使得对于任何元素 $x\in X$, $y\in Y$ 有 $x\le c\le y$.
\end{axiom}

\subsection{完备性公理与数集的上确界(下确界)的存在性}

\begin{definition}{上下界}
设集合 $X\subset \R$, 如果数 $c\in\R$ 存在, 使得对于任何 $x\in X$ 都有 $x\le c$, 就说集合 $X$ 是上有界集, 这时称 $c$ 为 $X$ 的上界.

同理可以定义下界, 既有上界又有下界的集合称为有界集.
\end{definition}

\begin{definition}{最值}
设 $a$ 是集合 $X\subset\R$ 的元素, 如果对于任何元素 $x\in X$ 都有 $x\le a$ 则元素 $a$ 称为集合 $X$ 的最大元素(最大值)记作 $\max X$ 或 $\max\limits_{x\in X} x$, 而最小值记作 $\min X$ 或 $\min\limits_{x\in X} x$.
\[
\begin{aligned}
	(a = \max X) := (a\in X \land \forall x \in X (x\le a)),\\
	(a = \min X) := (a\in X \land \forall x \in X (a\le x)).
\end{aligned}
\]
\end{definition}

\begin{definition}{上下确界}

集合 $X\subset\R$ 的上界中的最小者称为集合 $X$ 的上确界, 同理定义下确界如下
\[
\begin{aligned}
	(s = \sup X) &:= (\forall x\in X\, (x\le s))\land (\forall s' < s\,  \exists x' \in X (s'< x')),\\
	(i = \inf X) &:= (\forall x\in X\;(i\le x))\land (\forall i' > i\; \exists x' \in X (x'< i')).
\end{aligned}
\]
也有等价定义
\[
\begin{aligned}
	\sup X  := \min\{c\in\R\mid\forall x\in X(x\le c)\},\\
	\inf X  := \max\{c\in\R\mid\forall x\in X(c\le x)\}.
\end{aligned}
\]
\end{definition}

\begin{theorem}{上确界原理}{R1}
实数集的任何有上界非空子集有唯一的上确界.
\end{theorem}
\begin{idea*}
用完备性公理可证.	
\end{idea*}

\begin{proof}
	设 $X \subseteq \R$ 是给定的子集, 而 $Y = \{y\in\R | \forall x \in X (x \le y)\}$ 是 $X$ 的上界的集合, 依条件 $X\neq \varnothing,  Y\neq \varnothing$.于是, 根据完备性公理, 存在 $c \in \R$, 使得 $\forall x \in X, \forall y \in Y (x\le c\le y)$, 因此数 $c$ 既是 $X$ 的上界, 也是 $Y$ 的下届, 所以有 $c = \min Y = \max X$.
\end{proof}
\smallskip


在一些其他的实数公理系统中, Archimedes 原理常被列为公理, 不过我们实际上可以通过完备性公理证明此原理.

\begin{theorem}{Archimedes 原理}
	如果 $h$ 是任意一个固定的正数, 则对于任何实数 $x$, 可以找到唯一的整数 $k$, 使得 $(k-1)h\le x < kh$.
\end{theorem}
\begin{proof}
	因为整数集 $\Z$ 没有上界, 所以集合 $\{n\in\Z\mid x/h<n\}$ 是整数集的非空下有界子集.于是该集合有最小元素 $k$, 即 $(k-1) \le x/h < k$.因为 $h > 0$, 所以这些不等式等价于上述 Archimedes 原理中的不等式.由数集最小元素的唯一性可知, 满足上面两个不等式的整数 $k$ 是唯一的.
\end{proof}

根据几何公理 (在后面介绍几何时再详述), 对于一条直线 $\mathbb{L}$ 可以将其与实数集建立双射关系 $f:\mathbb{L} \to \R$.当上述双射 $f$ 存在时, 称直线 $\mathbb{L}$ 为坐标轴或数轴.

\begin{definition}{区间}
为了方便建立几何直观, 我们引入以下记号
\[
	\begin{aligned}
		(a, b) = \{a\in\R | a< x< b\},\\
		[a, b] = \{a\in\R | a\le x\le b\},\\
		(a, b] = \{a\in\R | a< x\le b\},\\
		[a, b) = \{a\in\R | a\le x< b\}.
	\end{aligned}
\]
其中 $(a, b)$ 是开区间, $[a, b]$ 是闭区间, $(a, b]$ 是左开右闭区间, $[a, b)$ 是左闭右开区间.对于一个区间 ab (无论其开闭)称 b-a 为其长度, 对于区间 $I$, 记其长度为 $\lvert I \rvert$.
\end{definition}

\begin{definition}{邻域}
包含点 $x \in \R$ 的开区间称为该点的邻域, 特别的, 当 $\delta > 0$ 时, 开区间 $(x - \delta, x+\delta)$ 称为点 $x$ 的 $\delta$ 邻域.
\end{definition}

\begin{definition}{模}
考虑数 $x, y\in\R$ 之间的距离, 也就是以它们为端点的区间的长度, 为了方便计算, 我们引入以下函数
\[
\begin{aligned}
	|\cdot|:\R &\to \R_{\ge 0}\\
	x &\mapsto \sqrt{x^2}.
\end{aligned}
\]
它称为一个数的模或者绝对值.称 $\lvert x - y\rvert$ 为 $x, y\in\R$ 之间的距离.
\end{definition}

% triangle inequality
\begin{theorem}{三角不等式}{ti}
根据模的定义, 我们可以得到三角不等式, 即对任何数 $x,y$, 不等式
\[
	\lvert x+y\rvert \le \lvert x\rvert + \lvert y\rvert.
\]
成立，并且等式当且仅当 $x$ 与 $y$ 同时非负或同时非正才成立.
\end{theorem}
\begin{proof}
如果 $0\le x$ 且 $0\le y$, 则 $0\le x+y$, $\lvert x+y\rvert = x+y$, $\lvert x \rvert = x$, $\lvert y \rvert = y$, 所以等式成立.

如果 $x \le 0$ 且 $y\le 0$, 则 $x+y \le 0$, $\lvert x+y\rvert = -(x+y) = -x-y$, $\lvert x \rvert = -x$, $\lvert y \rvert = -y$, 所以等式也成立.

如果 $x,y$ 一正一负, 例如 $x < 0 < y$, 则或者 $x < x+y \le 0$, 或者 $0 \le x+y <y$. 对于前者有 $\lvert x + y\rvert < \lvert x \rvert$, 对于后者有 $\lvert x + y\rvert < \lvert y \rvert$, 即对于二者都有 $\lvert x+y\rvert \le \lvert x\rvert + \lvert y\rvert$.
\end{proof}

\subsection{集合的基数}
\begin{definition}{基数}
集合 $A$ 的基数或者势, 记为 $|A|$ 或 $\mathrm{card}\ A$, 有限集的元素个数是一个自然数, 基数标志着该集合的大小. 对于有限集, 基数为其元素的数量.但对于有限集合大小并不是我们要研究的对象, Cantor 在建立集合论初就是为了研究无穷集合, 我们称自然数集
\[
	\mathbb{N}=\{1, 2, \cdots\}
\]
的基数为可数无穷, 记为 $\aleph_0$.
\end{definition}

在比较任意两个集合 $A$ 和 $B$ 的基数时, 我们有下面的定义: 
\begin{definition}
\ITEMS
	\item 我们称 $\lvert A \rvert \le \lvert B \rvert$, 如果存在从 $A$ 到 $B$ 的单射.
	\item 我们称 $\lvert A \rvert = \lvert B \rvert$, 如果$\lvert A \rvert \le \lvert B \rvert$ 且 $\lvert B \rvert \le \lvert A \rvert$, 也就是说, 存在从 $A$ 到 $B$ 的双射.
\ITEME
\end{definition}

\begin{theorem}{无最大基数定理}
对于任一非空集合 $A$, 我们总有 $\lvert A \rvert<\lvert \mathcal{P}(A) \rvert$.
\end{theorem}

\begin{proof}

我们先证明 $\lvert A \rvert \le \lvert \mathcal{P}(A) \rvert$\\
定义映射 $g:A\rightarrow \mathcal{P}(A)$ 如下
\[ 
	g(x)=\{x\},  \forall x \in A.
\] 
不难看出, $g$ 是单射.

下面我们用反证法证明 $\lvert A \rvert \ne \lvert \mathcal{P}(A) \rvert$:
若相等, 则存在双射
\[
	f:A\rightarrow \mathcal{P}(A).
\]
定义一个$A$的子集合$B$如下: 
\[
B = \{x\in A: x\notin f(x)\}.
\]
由于 $f$ 是双射, 所以必存在(唯一)元素 $y\in A$ 使得
\[
f(y) = B.
\]
考虑 $y$ 与 $B$ 的关系, 发现无论如何都会产生矛盾: 
\ITEMS
	\item 如果 $y\in B=f(y)$,  由 $B$ 的定义, 我们会得到 $y\notin B$；
	\item 如果 $y\notin B=f(y)$,  由 $B$ 的定义, 我们会得到 $y\in B$.
\ITEME

我们记 $\aleph_n=2^{\aleph_{n-1}},  n=1,  2,  \cdots$.由上述定理, 显然我们有 
\[
	\aleph_0 < \aleph_1 < \aleph_2 <\cdots< \aleph_n< \cdots.
\]
\end{proof}

\begin{definition}{连续统的势}

\begin{theorem}{Cantor 定理}
连续统的势大于自然数集的势, 即
\[
\lvert \N \rvert < \lvert \R \rvert.
\]
\end{theorem}

\begin{proof} $c = \aleph_1$

我们先证明 $\lvert \R \rvert=\lvert (0, 1]\rvert $
\smallskip
\ITEMS
\item $\lvert \R\rvert =\lvert (-1, 1)\rvert $, 不难看出映射  $f:(-1, 1)\rightarrow \R$ 满足 $f(x)=\frac {x}{1-x^2}$ 是双射.

\item $\lvert (-1, 1)\rvert=\lvert (0, 1)\rvert $, 不难看出映射  $f:(-1, 1)\rightarrow (0, 1) $ 满足 $f(x)=\frac {x+1}{2}$ 是双射.

\item $\lvert (0, 1)\rvert=\lvert (0, 1]\rvert $, 因为 $(0, 1)\subseteq (0, 1]\subseteq (-1, 1)$.
\ITEME

\smallskip
下面我们证明 $\lvert (0, 1]\rvert=2^{\aleph_0}$.

把 $(0, 1]$ 中的任一实数 $x$ 用二进制表示: 更具体地讲, 就是找到每项取值是 $0$ 或 $1$ 的无穷数列 $\{a_n\}_{n \ge 1}$ 满足在二进制下 $x=0.a_1 a_2\cdots a_n \cdots$, 其中无穷多个 $a_n$ 是 $1$ .
\ITEMS 
	\item 第一步: 把区间 $(0, 1]$ 二等分为两半开闭子区间: 若 $x$ 属于靠左的(称为第一个, 后同)子区间 $(0, \frac{1}{2}]$, 则 $a_1=0$；若 $x$ 属于靠右的(称为第二个, 后同)子区间, 则 $a_1=1$；
	
	\item 第二步: 把 $x$ 所属的第一步中的小区间二等分为两半开闭子区间:  若 $x$ 属于第一个子区间, 则 $a_2=0$；若 $x$ 属于第二个子区间, 则 $a_2=1$；
	\item $\vdots$ 
	\item 第 $n$ 步:  把 $x$ 所属的上一步中的小区间二等分为两半开闭子区间:  若 $x$ 属于第一个子区间, 则 $a_n=0$；若 $x$ 属于第二个子区间, 则 $a_n=1$；
	\item $\vdots$
\ITEME

注意到上述 $a_n$ 中, 有无穷多个是 $1$.这样我们就建立了 $(0, 1]$ 和集合 
\[
    S=\{(a_1,  a_2, \cdots,  a_n,  \cdots): \mbox{每个 $a_n$ 是 $0$ 或 $1$ 且 其中无穷多个 $a_n$ 是 $1$} \}
\] 
的一一对应关系.

决定 $\lvert S\rvert$, 令 
\[
	T=\{(a_1,  a_2, \cdots,  a_n,  \cdots): \mbox{每个 $a_n$ 是 $0$ 或 $1$ 且 其中只有限多个 $a_n$ 是 $1$} \}
\] 
和 
\[
	R=\{(a_1,  a_2, \cdots,  a_n,  \cdots): \mbox{每个 $a_n$ 是 $0$ 或 $1$}\}.
\]

不难看出 $\lvert R\rvert=2^{\aleph_0},  \lvert T\rvert=\aleph_0$ 且 $S\cup T= R$.由上面关于基数的基本定理 $2$, 我们有 
\[
	\lvert S\rvert=\lvert S\cup T\rvert =\lvert R\rvert=2^{\aleph_0}=\aleph_1.
\]
\end{proof}
\smallskip

\begin{proposition}{连续统假设}
不存在一个无穷集合, 其基数严格介于自然数集的基数 $\aleph_0$ 与实数集的基数 $\aleph_1$(即连续统的基数)之间. 即不存在一个无穷集合, 其基数严格介于 $\aleph_n$ 与 $\aleph_{n+1},  n=0,  1, 2,  \cdots $ 之间.
\end{proposition}

连续统假设是一个在 ZFC 公理化集合论体系内不可证(不能被证明、也不能被证伪)的命题.

\section{关于实数集完备性的一些基本定理}

\subsection{闭区间套原理 (Cauchy-Cantor 原理)}

\begin{definition}{集合套}
设 $X_1, X_2, \cdots, X_n, \cdots$ 是某些闭区间的序列, 如果 $X_1\supset X_2\supset \cdots \supset X_n \supset\cdots$, 且对 $\forall \varepsilon > 0$, 在序列中可以找到长度 $|X_k| < \varepsilon$ 的区间, 我们称 $X_1, X_2, \cdots, X_n, \cdots$ 为闭区间套序列.
\end{definition}

\begin{theorem}{Cauchy-Cantor 原理}{R2}
	对于任何闭区间的渐降列 $I_1\supset I_2\supset \cdots \supset I_n \supset\cdots$, 可以找到一点 $c\in\R$ 属于所有这些闭区间.如果除此之外还已知该序列为闭区间套序列, 则 $c$ 是所有闭区间的唯一公共点.
\end{theorem}

\begin{idea*}
使用完备性定理可证.	
\end{idea*}

\begin{proof}
首先	指出, 对于上述序列中的任何两个闭区间 $I_m = [a_m, b_m], I_n = [a_n, b_n]$ 必有 $a_m \le b_n$.因此, 对于数集 $A =\{a_m | m\in \N\},  B= \{b_n| n\in \N\}$, 根据完备性公理, 可以找到数 $c\in\R$ 使得对于 $\forall a_m\in A, \forall b_n\in B$ 有 $a_m\le c\le b_n$. 特别的, 对于 $\forall n\in N$ 都有 $a_n\le c \le b_n$, 即点 $c$ 属于所有闭区间 $I_n$.

现在, 假设有两个不同的有相似性质的点 $c_1, c_2$, 假如 $c_1 < c_2$, 则对于任何 $n\in\N$ 均有,  $a_n \le c_1 < c_2 \le b_n$ 从而 $0 < c_1 - c_2 < b_n - a_n$. 于是, 上述序列中的每一个闭区间的长度不可能小于 $c_2 - c_1 > 0$, 这意味着, 如果在该序列中有长度任意小的闭区间则公共点是唯一的. 
\end{proof}
\smallskip

\subsection{有限覆盖原理 (Borel–Lebesgue 原理, Heine-Borel 原理)}

\begin{definition}{覆盖}
设 $S=\{X\}$ 是由一组集合 $X$ 构成的集合族, 如果 $Y\subset \bigcup\limits_{X\in S}X$, 那么我们说 $S$ 覆盖集合 $Y$.
\end{definition}

集合族 $S = {X}$ 的子集也是一个集合组, 称为 $S$ 的子族.因此, 集合族的子族本身是同一类型的集合族.

\begin{theorem}{Borel–Lebesgue 原理}{R3}
在覆盖一个闭区间的任何开区间族中都有覆盖该闭区间的有限子族.
\end{theorem}
\begin{proof}
设 $S =\{U\}$ 是覆盖闭区间 $[a, b] = I_1$ 的开区间族, $U$ 是构成 $S$ 的开区间.假如闭区间 $I_1$ 不能被 $S$ 中有限个开区间覆盖, 我们将闭区间 $I_1$ 等分为两个闭区间, 则其中至少有一个不能被 $S$ 中有限个开区间覆盖, 我们把这个区间记为 $I_2$, 我们再通过类似的方法得到 $I_3, I_4, \cdots$.

于是我们就构建出了一个闭区间套序列 $I_1 \supset I_2 \supset I_3 \supset \cdots$, 而且根据其构造方法可知, 第 $n$ 个区间的长度为 $\lvert I_n \rvert = \frac{\lvert I_1 \rvert}{2^{n-1}}$, 所以在此序列中有长度任意小的闭区间, 根据 \hyperref[thm:R2]{闭区间套原理}, 存在一点 $c$ 包含于所有闭区间 $I_n$, 而考虑 $c$ 点的一 $\varepsilon$ 邻域 $U = (c - \varepsilon,  c+\varepsilon)$, 在上述闭区间序列中总能找出一个 $I_n$ 使得 $\lvert I_n \rvert < \varepsilon$, 所以有 $I_n \subset U$, 这与 $I_n$ 不能被开区间族的有限子族覆盖相矛盾.
\end{proof}
\smallskip

\subsection{极限点定理 (Bolzano-Weierstrass 原理)}
极限点定理也称为致密性定理, 其有另一种表述将在后文介绍 (\hyperref[thm:R4']{致密性定理}) .

\begin{definition}{极限点}
如果点 $p\in\R$ 的任何邻域都包含集合 $X\subset\R$ 的一个无穷子集, 点 $p$ 就称为集合 $X$ 的极限点.
\end{definition}

\begin{theorem}{Bolzano-Weierstrass 原理}{R4}
任何无穷有界数集至少有一个极限点.
\end{theorem}
\begin{proof}
设 $X$ 是 $\R$ 的给定子集并且满足定理条件, 由有界集的定义可知, 某闭区间 $I = [a, b] \subset\R$ 包含集合 $X$. 我们证明集合 $I$ 上至少有一个点是 $X$ 的极限点.

假如并非如此, 每个点 $x \in I$ 就会有这样的邻域 $U(x)$, 其中或者根本没有 $X$ 中的点, 或者只有有限个属于集合 $X$ 的点, 对每个点 $x\in I$ 构造出的这种邻域的总体 $\{U\}$ 组成闭区间 $I$ 的开区间覆盖, 根据 \hyperref[thm:R3]{有限覆盖原理}, 可以从上述覆盖中, 选出有限个开区间 $I(x_1), \cdots, U(x_n)$, 使它们仍覆盖区间 $I$.因为 $X \subset I$, 所以它们也覆盖集合 $X$.然而 $X$ 并不是有限集, 我们得到了矛盾.
\end{proof}

\chapter{极限论}

在讨论初等数学与高等数学的区别时, 极限思想往往被认为是二者之间最根本的区别, 也是分析学的基础. 初等数学侧重于 ``常量" 与 ``静态" 的计算, 高等数学则侧重于 "变量" 与 ``动态" 的变化过程，而极限正是研究动态问题的最基本手段。

事实上, 极限思想很早就在数学家心中萌生. 无论是刘徽的割圆术, 还是芝诺的阿基里斯追龟悖论, 都体现了极限思想; 只不过, 在 Cauchy 系统建立极限理论之前的两千多年间, 数学界长期未能给出普遍接受的严格定义。

\section{数列极限与性质}
在前面我们介绍了序列的定义.

\begin{definition}{数列}
若序列的值域为实数集或其子集 $f:\N \to \R$, 我们称该序列为数列 (Sequence).	
\end{definition}


\subsection{数列极限的定义}

我在这里介绍两种等价的定义, 首先介绍根据邻域的思想(拓扑)所给出的定义.

\begin{definition}{数列极限邻域定义}
	数 $A \in \R$ 称为数列 $\{x_n\}$ 的极限, 如果对于点 $A$ 的任何一个邻域 $V(A)$, 都存在序号 $N$, 使得数列中所有序号大于 $N$ 的项都包含在点 $A$ 的上述邻域 $V(A)$ 中, 即
\[
	\left( \lim_{n\to\infty} x_n = A\right) := \forall V(A)\ \exists N\in\N\ \forall n > N (x_n \in V(A)).
\]
\end{definition}

接下来指出数列极限定义的另一种常见表述 (Cauchy, D’Alembert,).

\begin{definition}{数列极限的 $\varepsilon - N$}
数 $A \in \R$ 称为数列 $\{x_n\}$ 的极限, 如果对于任何 $\varepsilon >0$ 都存在序号 $N$, 使得对于一切 $n>N$, 都有 $\lvert x_n - A \rvert < \varepsilon$, 即
\[
	\left( \lim_{n\to\infty} x_n = A\right) := \forall \varepsilon > 0\ \exists N\in\N\ \forall n > N (\lvert x_n - A\rvert < \varepsilon).
\]

如果 $\lim\limits_{n\to\infty} x_n = A$, 我们就说数列 $\{x_n\}$ 收敛于 $A$ 或趋于 $A$, 记作 $n\to\infty$ 时 $x_n \to A$. 有极限的数列称为收敛 (Convergent) 数列, 没有极限的数列称为发散 (Divergent) 数列. 
\end{definition}


\subsection{收敛数列的性质}
接下来我们研究收敛数列所拥有的分析性质, 不过未来将看到这些性质并非数列所所特有的. 

\begin{definition}{常数列}
	只取一个值的数列为常数列, 数列 $\{x_n\}$ 称为最终常数列, 如果存在数 $A$ 与序号 $N$, 使得对于任何 $n > N$ 均有 $x_n = A$.
\end{definition}

\begin{definition}{有界数列}
	数列 $\{x_n\}$ 称为有界数列, 如果存在数 $M$, 使得对于任何 $n \in N$ 均有 $x_n < M$.
\end{definition}

数列极限有如下性质

\begin{theorem}
最终常数列收敛.
\end{theorem}

\begin{theorem}
数列极限的任何一个邻域都包含数列中为数有限的项之外的所有项.
\end{theorem}

以上两条性质皆由定义显然. 而接下来介绍一些更具一般性的性质.

% 唯一性
\begin{theorem}{数列极限的唯一性}{Slimit_p1}
数列不可能有两个不同的极限.
\end{theorem}
\begin{idea*}
使用反证法, 假设数列有两个不同的极限, 证明两个极限相同.
\end{idea*}

\begin{proof}
	设 $\lim\limits_{n\to\infty} x_n = A_1 , \ \lim\limits_{n\to\infty} x_n = A_2$.假如 $A_1 \neq A_2$, 我们就选定点 $A_1, A_2$ 的两个不相交的邻域 $V(A_1), V(A_2)$.
	
	例如, 可以在 $\delta < \frac{\lvert A_1 - A_2\rvert}{2}$ 时取这两个点的邻域, 显然 $V(A_1), V(A_2)$ 不相交. 根据极限的定义, 可以求得 $N_1, N_2$ 使得 $\forall n > N_1 (x_n\in V(A_1)), \ \forall n > N_2 (x_n\in V(A_2))$.于是对于 $n > \max\{N_1, N_2\}$ 有 $x_n \in V(A_1) \cap V(A_2)$, 与 $V(A_1), V(A_2)$ 不相交矛盾.
\end{proof}
\smallskip

% 有界性
\begin{theorem}{数列极限的有界性}{Slimit_p2}
收敛数列有界.
\end{theorem}

\begin{idea*}
根据极限定义对于 $N$ 之后的值显然有 $A - \epsilon < x_n < A + \epsilon$, 那么显然有限数列是有界的.
\end{idea*}

\begin{proof}
	设 $\lim\limits_{n\to\infty} x_n = A$.如果取 $\varepsilon = 1$, 就可以求出 $N$, 使得 $\forall n > N (\lvert x_n - A \rvert < 1)$. 这意味着, 当 $n > N$ 时有 $\lvert x_n \rvert < \lvert A \rvert + 1$.如果现在取
	\[
		M > \max\{\lvert x_1\rvert, \lvert x_2\rvert, \cdots, \lvert x_N\rvert, \lvert A\rvert + 1\},
	\]
	就得到 $\forall n (x_n < M)$.
\end{proof}
\smallskip

% 保序性
\begin{theorem}{数列极限的保序性}{Slimit_p3}
设有收敛数列 $\{x_n\},\{y_n\}$, 满足 $\lim\limits_{n\to\infty} x_n = A, \lim\limits_{n\to\infty} y_n = B$ 且有 $a < b$, 则存在 $N \in \N^+$ 使得对任意 $n > N$ 有
\[
	x_n < y_n.
\]

\end{theorem}
\begin{idea*}
设 $\varepsilon = \frac{B - A}{2}$, 则有 $N$ 使得 $x_n < \frac{B + A}{2},\ y_n > \frac{B + A}{2}$.
\end{idea*}
\begin{proof}
	取数 $C$, 使得 $A < C < B$. 根据极限的定义, 可以求出序号 $N'$ 和 $N''$, 使得对于任何 $n>N'$ 都有 $\lvert x_n - A \rvert < C - A$, 并且对于任何 $n > N \in \N$, 都有 $\lvert y_n - B \rvert < B - C$. 于是, 当 $n>N = \max\{N',N''\}$ 时, 就得到 $x_n < A + (C - A) = C = B - (B- C) < y_n$.
\end{proof}
\smallskip

% 保号性
\begin{corollary}{数列极限的保号性}{Slimit_p4}
	设 $\lim\limits_{n\to\infty} x_n = A \neq 0$, 则存在 $N \in \N^+$, 当 $n > N$ 时有
	\ITEMS
		\item 若 $A > 0$,
		\[
			x_n > \frac A2 > 0.
		\]
		\item 若 $A < 0$,
		\[
			x_n < \frac A2 < 0.
		\]
	\ITEME
\end{corollary}

保号性由保序性显然.
\smallskip

% 迫敛性
\begin{theorem}{数列极限的迫敛性 (夹逼定理)}{Ssandwich}
设数列 $\{x_n\}, \{y_n\}, \{z_n\}$ 满足以下条件, 对于任何 $n > N \in \N$, 关系式 $x_n \le y_n \le z_n$ 成立.如果数列 $\{x_n\}, \{z_n\}$ 收敛于同一个极限, 则数列 $\{y_n\}$ 也收敛于该极限. 
\end{theorem}

\begin{idea*}
由定义显然，对于任意 $\varepsilon$ 只需取最大的 $N$ 即可.
\end{idea*}

\begin{proof}
设 $\lim\limits_{n\to\infty} x_n = \lim\limits_{n\to\infty} z_n = A$. 由 $\varepsilon > 0$ 求出序号 $N'$ 和 $N''$, 使得对于任何 $n>N'$ 都有 $N'< A + \varepsilon$. 于是, 当 $n > N =\max\{N',N''\}$ 时, 我们得到 $A -\varepsilon < x_n \le y_n \le z_n < A + \varepsilon$, 即 $\lvert y_n - A\rvert < \varepsilon$, 从而 $\lim\limits_{n\to\infty} y_n = A$.
\end{proof}
\smallskip

\begin{corollary}
	设 $\lim\limits_{n\to\infty} x_n = A,  \lim\limits_{n\to\infty} y_n = B$.如果存在序号 $N$, 是的对于任何 $n > N$ 都有
	\ITEMS
		\item $x_n > y_n$, 则 $A\ge B$.
		\item $x_n \ge y_n$, 则 $A\ge B$.
		\item $x_n > B$, 则 $A\ge B$.
		\item $x_n \ge B$, 则 $A\ge B$.
	\ITEME
\end{corollary}

除了上述性质外, 我们也需要讨论在极限过程中代数运算的性质.  

% 运算性质
\begin{theorem}{极限的四则运算}{Slimit_p5}
设 $\{x_n\}, \{y_n\}$ 是两个数列, $\alpha, \beta$ 是两个常数, 如果 $\lim\limits_{n\to\infty} x_n = A, \ \lim\limits_{n\to\infty} y_n = B$ 则
	\ITEMS
		\item $\lim\limits_{n\to\infty} (\alpha x_n + \beta y_n) = \alpha A + \beta B$.
		\item $\lim\limits_{n\to\infty} x_n \cdot y_n = A \cdot B$.
		\item 当 $y_n\neq 0, \ n\in\N$ 且 $B \neq 0$ 时 $\lim\limits_{n\to\infty} \frac{x_n}{y_n} = \frac AB$.
	\ITEME
\end{theorem}
 
\begin{idea*}
	构造性证明, 通过换元 $A = x_n + a,\ B = y_n + b$ 与不等式关系使得原式小于任意给定的 $\varepsilon$.
\end{idea*}

\begin{proof}
	我们逐个来证明:
	\ITEMS
		\item 设给定 $\varepsilon > 0$, 对于数列 $\{x_n\}$ 存在 $N'$ 使得 $\forall n > N'$ 有 $\lvert x_n - A \rvert < \frac{\varepsilon}{2 \alpha}$. 类似的, 对于数列 $\{y_n\}$ 存在 $N''$ 使得 $\forall n > N''$ 有 $\lvert y_n - B \rvert < \frac{\varepsilon}{2 \beta}$. 于是, 当 $n > \max\{N', N''\}$ 时, 有
		\[
			\lvert(\alpha A + \beta B) - (\alpha x_n + \beta y_n)\rvert \le \lvert \alpha A - \alpha x_n\rvert +\lvert \beta B - \beta y_n\rvert < \frac{\varepsilon}{2}+\frac{\varepsilon}{2} =\varepsilon.
		\]
		
		\item 注意到
		\[
		\begin{aligned}
			\lvert x_n y_n - AB\rvert &= \lvert [A + (x_n - A)][B + (y_n - B)] - AB \rvert\\ 
			&= \lvert AB + B(x_n - A) + A(y_n - B) + (x_n - A)(y_n - B) - AB  \rvert \\
			&= \lvert B(x_n - A) + A(y_n - B) + (x_n - A)(y_n - B) \rvert \\
			&\le \lvert A\rvert \lvert y_n - B\rvert + \lvert B\rvert \lvert x_n - A\rvert + \lvert x_n - A \rvert \lvert y_n - B \rvert
		\end{aligned}
		\]
		
		对于任意给定的 $\varepsilon > 0$, 根据极限定义存在 $N',N''$, 使得
		\[
		\begin{aligned}
			\forall n > N'\left(\lvert x_n - A\rvert < \min\left\{1,\frac{\varepsilon}{3(\lvert B\rvert + 1)}\right\} \right),\\
			\forall n > N''\left(\lvert y_n - B\rvert < \min\left\{1,\frac{\varepsilon}{3(\lvert A\rvert + 1)}\right\} \right).
		\end{aligned}
		\]
		则对于 $n > \max\{N',N''\}$ 有
		\[
			\lvert x_n \rvert \le \lvert A \rvert + \lvert x_n - A \rvert < \lvert A \rvert + 1,
		\]
		\[
			\lvert y_n \rvert \le \lvert B \rvert + \lvert y_n - B \rvert < \lvert B \rvert + 1,
		\]
		\[
			\lvert x_n - A \rvert \lvert y_n - B \rvert < \min\left\{1,\frac \varepsilon3\right\}^2 \le \frac\varepsilon3,
		\]
		由此可得
		\[
			\lvert x_n \rvert \lvert y_n - B \rvert < (\lvert A \rvert + 1) \frac{\varepsilon}{3(\lvert A\rvert + 1)} < \frac{\varepsilon}{3},
		\]
		\[
			\lvert y_n \rvert \lvert x_n - A \rvert (\lvert B \rvert + 1) \frac{\varepsilon}{3(\lvert B\rvert + 1)} < \frac{\varepsilon}{3},
		\]
		\[
			\lvert x_n - A \rvert \lvert y_n - B \rvert < \frac{\varepsilon}{3}.
		\]
		即 $\lvert x_n y_n - AB\rvert < \varepsilon$.
		
		\item 注意到
		\[
		\begin{aligned}
			\left\lvert \frac{A}{B} - \frac{x_n}{y_n}\right\rvert 
				&= \left\lvert \frac{A y_n - B x_n}{B y_n} \right\rvert\\
				&= \left\lvert \frac{(x_n + A - x_n) y_n - (y_n + B - y_n) x_n}{(y_n + B - y_n) y_n} \right\rvert\\
				&= \left\lvert \frac{x_n y_n + (A - x_n) y_n - x_n y_n - (B - y_n) x_n}{\left(1 + \frac{B - y_n}{y_n}\right) y_n^2} \right\rvert\\
				&= \left\lvert \frac{(A - x_n) y_n - (B - y_n) x_n}{y_n^2} \right\rvert \left\lvert \frac{1}{1 + \frac{B - y_n}{y_n}} \right\rvert\\
				&\le \frac{\lvert A - x_n\rvert \lvert y_n \rvert + \lvert B - y_n \rvert \lvert x_n \rvert}{y_n^2} \frac{1}{1 - \frac{\lvert B - y_n \rvert}{\lvert y_n \rvert}}.
		\end{aligned}
		\]
		
		对于任意给定的 $\varepsilon > 0$, 根据极限定义存在 $N',N''$, 使得
		\[
			\forall n > N'\left(\lvert x_n - A\rvert < \min\left\{1,\frac{\varepsilon\lvert B \rvert}{8}\right\} \right),
		\]
		\[
			\forall n > N''\left(\lvert y_n - B\rvert < \min\left\{ \frac{\lvert B\rvert}{4},\frac{\varepsilon B^2}{16(\lvert A\rvert + 1)}\right\} \right).
		\]
		由 $\lim\limits y_n=B\neq 0$, 对于 $n > \max\{N',N''\}$ 有
		\[
			\lvert x_n \rvert \le \lvert A \rvert + \lvert x_n - A \rvert < \lvert A \rvert + 1,
		\]
		\[
			\lvert y_n \rvert \ge \lvert B \rvert - \lvert y_n - B \rvert > \lvert B \rvert - \frac{\lvert B \rvert}{4} > \frac{\lvert B \rvert}{2},
		\]
		\[
			0 \le \frac{\lvert B - y_n \rvert}{\lvert y_n \rvert} < \frac{\lvert B \rvert / 4}{\lvert B \rvert / 2} = \frac12
		\]
		\[
			1 - \frac{\lvert B - y_n \rvert}{\lvert y_n \rvert} > \frac{1}{2},
		\]
		由此可得
		\[
			\frac{\lvert A - x_n\rvert}{\lvert y_n \rvert} < \frac{2}{\lvert B \rvert} \frac{\varepsilon \lvert B \rvert}{8} = \frac{\varepsilon}{4}
		\]
		\[
			\frac{\lvert B - y_n \rvert \lvert x_n \rvert}{y_n^2} < (\lvert A \rvert + 1) \frac{4}{B^2} \frac{\varepsilon B^2}{16(\lvert A\rvert + 1)} = \frac{\varepsilon}{4} = \frac{\varepsilon}{4}
		\]
		\[
			0 < \frac{1}{1 - \frac{\lvert B - y_n \rvert}{\lvert y_n \rvert}} < 2
		\]
		即 $\left\lvert \frac{A}{B} - \frac{x_n}{y_n}\right\rvert  < \varepsilon$.

	\ITEME
\end{proof}
\smallskip

\begin{remark}
	以上是构造性的证明方式, 也常见一种非构造性的分析性与归纳性证明, 可以参考陈纪修老师的数学分析. 在数学构成主义（Constructivism）看来, 证明一个对象存在, 就必须把它构造出来. 他们拒绝使用排中律等非构造性逻辑. 不过两者都属于 $\varepsilon - N$ 语言下的严格证明.
\end{remark}

\subsection{无穷小与无穷大}
\begin{definition}{无穷小量}
	称极限值为 $0$ 的数列为无穷小量.
\end{definition}
\begin{remark}
无穷小量不是某个固定的很小的数, 而是一个变量 (数列).	
\end{remark}

由无穷小量的定义我们可以直接得到
\[
	\lim_{n \to \infty} x_n = a \iff \{x_n - a\} \text{是无穷小量}.
\]

\begin{definition}{无穷大量}
	若对于任意给定的 $G > 0$, 可以找到正整数 $N$, 使得当 $n > N$ 时成立
	\[
		\lvert x_n \rvert > G,
	\] 
	则称数列 $\{x_n\}$ 是无穷大量, 记为
	\[
		\lim_{n\to\infty} x_n = \infty.
	\]
	特别的, 如果 $\{x_n\}$ 从某一项开始都是正 (负) 的, 则称其为 正 (负) 无穷大量, 分别记为
	\[
		\lim_{n\to\infty} x_n = +\infty (\lim_{n\to\infty} x_n = -\infty)
	\]	
\end{definition}

\begin{theorem}{无穷大量与无穷小量的关系}{inftyrelation}
	设 $x_n \neq 0$, 则 $\{x_n\}$ 是无穷大量的充要条件是 $\left\{\frac{1}{x_n}\right\}$ 是无穷小量.
\end{theorem}

\begin{proof}
	有 $\{x_n\}$ 是无穷大量, 对于 $\forall \varepsilon > 0$, 取 $G = \frac{1}{\varepsilon} > 0$, 由无穷大量的定义, $\exists N, \forall n > N: \lvert x_n \rvert > G = \frac{1}{\varepsilon}$ 则有 $\left\lvert\frac{1}{x_n} \right\rvert< \varepsilon$, 即 $\left\{\frac{1}{x_n}\right\}$ 是无穷小量.
	
	同理可证充分性.
\end{proof}

% Stolz定理
\subsection{Stolz 定理}

我们在处理无穷大量与无穷小量的商的极限问题时, 常常会遇到形如 $\infty/\infty$ 或 $0/0$ 的极限, 其结果可能是无穷大量或实数极限, 但也有可能不存在极限, 我们称这种极限为不定式或待定式 (Indeterminate form).

\begin{remark}
虽然也有其他的形如 $0\cdot \infty , 1^{\infty }, \infty -\infty , 0^{0}, \infty ^{0}$ 等不定式, 但总是可以将其化为 $\infty/\infty$ 或 $0/0$ 的不定式. 实际上 $\infty/\infty$ 型或 $0/0$ 型极限本质是相同的. 但是针对这两种情形, Stolz 定理的内容有一些细微的差异.
\end{remark}


\begin{theorem}{Stolz 定理 ($\infty / \infty$ 型)}{stolz_1}
	设有两数列 $\{x_n\}, \{y_n\}$, 其中 $\{y_n\}$ 是严格单调递增数列, 有 $y_n \to +\infty$, 且
	\[
		\lim_{n\to\infty} \frac{x_n - x_{n-1}}{y_n - y_{n-1}} = a
	\]
	其中 $a \in \R \cup \{\pm \infty\}$, 则
	\[
		\lim_{n\to\infty} \frac{x_n}{y_n} = a.
	\]
\end{theorem}

\begin{idea*}
	对 $a$ 的取值进行分类讨论, 先证明 $a = 0$ 的情况, 那么其他情况也就迎刃而解了. 而 $a = 0$ 的情况主要想法是用三角不等式将数列 $\left\{\frac{x_n}{y_n}\right\}$ 分成两部分, 并使其分别趋于 $0$ 即可证明, 难点在于如何拆分该数列.
\end{idea*}

\begin{proof}
	\ITEMS
		\item 当 $a = 0$ 时, 由 $\lim\limits_{n\to\infty} \frac{x_n - x_{n-1}}{y_n - y_{n-1}} = a$ 可知 $\forall \varepsilon > 0,\ \exists N',\ \forall n > N'$ 有
		\[
			\lvert x_n - x_{n-1} \rvert < \frac{\varepsilon}{2} (y_n - y_{n-1})
		\]
		由于 $y_n \to +\infty$, 我们可以要求 $y_{N'} > 0$, 又因为 $y_n$ 是严格单调递增数列有
		\[
			\lvert x_n - x_{n-1} \rvert + \lvert x_{n-1} - x_{n-2} \rvert + \cdots + \lvert x_{N'+1} - x_{N'} \rvert < \frac{\varepsilon}{2} (y_n - y_{n-1}) + \frac{\varepsilon}{2} (y_{n-1} - y_{n-2}) + \cdots + \frac{\varepsilon}{2} (y_{N'+1} - y_{N'}),
		\]
		由三角不等式化简可得
		\[
			\lvert x_n - x_{N'} \rvert < \frac{\varepsilon}{2} (y_{n} - y_{N'}).
		\]
		对不等式两边同时除以 $y_n > 0$ 则有
		\[
			\left\lvert \frac{x_n}{y_n} - \frac{x_{N'}}{y_n} \right\rvert < \frac{\varepsilon}{2} \left(1 - \frac{y_{N'}}{y_n}\right) < \frac{\varepsilon}{2}.
		\]
		由 $y_n \to +\infty$ 可知 $\exists N'' > N'$ 使得 $\forall n > N''$ 时有
		\[
			\left\lvert \frac{x_{N'}}{y_n} \right\rvert < \frac{\varepsilon}{2}.
		\]
		取 $N = \max\{N',\ N''\}$, 对于 $\forall n > N$ 综上可得
		\[
			\left\lvert \frac{x_n}{y_n} \right\rvert \le \left\lvert \frac{x_n}{y_n}- \frac{x_{N'}}{y_n} + \frac{x_{N'}}{y_n} \right\rvert \le \left\lvert \frac{x_n}{y_n}- \frac{x_{N'}}{y_n}\right\rvert + \left\lvert \frac{x_{N'}}{y_n}\right\rvert < \frac{\varepsilon}{2} + \frac{\varepsilon}{2} = \varepsilon.
		\]
		即 $\lim\limits_{n\to\infty} \frac{x_n}{y_n} = 0$.
		\smallskip
		
		\item 当 $a \in \R\setminus\{0\}$ 时, 基本想法是构造数列使得我们可以应用 $a = 0$ 情况下的结论. 于是便想到如下构造
		\[
			x_n' = x_n - a y_n.
		\]
		于是有
		\[
			\lim_{n\to\infty} \frac{x_n' - x_{n-1}'}{y_n - y_{n-1}} = \lim_{n\to\infty} \frac{x_n - a y_n - x_{n-1} + a y_{n-1}}{y_n - y_{n-1}} = \lim_{n\to\infty} \frac{x_n - x_{n-1}}{y_n - y_{n-1}} - a = 0,
		\]
		显然由 $a = 0$ 情况下的结论可知 $\lim\limits_{n\to\infty} \frac{x_n'}{y_n} = 0$ 从而
		\[
			\lim_{n\to\infty} \frac{x_n}{y_n} = a.
		\]
		\smallskip
		
		\item 当 $a = +\infty$ 时, 存在 $N > 0,\ \forall n > N$ 有
		\[
			x_n - x_{n-1} > y_n - y_{n-1},
		\] 
		则可知 $\{x_n\}_{n>N}$ 也是严格单调递增数列, 且可知 $\{x_n\}$ 为正无穷大量. 由于
		\[
			\lim_{n \to \infty} \frac{x_n -x_{n-1}}{y_n - y_{n-1}} = +\infty.
		\]
		由定理 \hyperref[thm:inftyrelation]{定理 3.8} 可得
		\[
			\lim_{n \to \infty} \frac{y_n - y_{n-1}}{x_n -x_{n-1}} = 0,
		\]
		那么我们可以使用 $a = 0$ 情况下的结论得到
		\[
			\lim_{n \to \infty} \frac{y_n}{x_n} = 0,
		\]
		再由 \hyperref[thm:inftyrelation]{定理 3.8} 有
		\[
			\lim_{n \to \infty} \frac{x_n}{y_n} = +\infty.
		\]
		\smallskip
		
		\item 当 $a = -\infty$ 时, 注意到若令 $x_n' = -x_n$ 则可变为 $a = +\infty$ 的情况, 则可得
		\[
			\lim_{n \to \infty} \frac{x_n}{y_n} = -\infty.
		\]
		
	\ITEME
\end{proof}
\smallskip

\begin{remark}
假若 $a = \infty$ 而非定号无穷大, 那么 Stolz 定理不成立.
\end{remark}
\smallskip

\begin{theorem}{Stolz 定理 ($0/0$ 型)}{stolz_2}
	设有两数列 $\{x_n\}, \{y_n\}$, 其中 $\{y_n\}$ 是严格单调递增数列, 有 $x_n \to 0,\ y_n \to 0$, 且
	\[
		\lim_{n\to\infty} \frac{x_n - x_{n+1}}{y_n - y_{n+1}} = a
	\]
	其中 $a \in \R \cup \{\pm \infty\}$, 则
	\[
		\lim_{n\to\infty} \frac{x_n}{y_n} = a.
	\]
\end{theorem}

\begin{proof}
	\ITEMS
		\item 当 $a \in \R$ 时, 由极限定义有, 对于 $\forall \varepsilon > 0$, 存在 $N \in \N$ 使得 $n > N$ 时有
		\[
			a - \varepsilon < \frac{x_n - x_{n+1}}{y_n - y_{n+1}} < a + \varepsilon,
		\]
		则有
		\[
			(a - \varepsilon)(y_n - y_{n+1}) < x_n - x_{n+1} < (a + \varepsilon)(y_n - y_{n+1}),
		\]
		进一步可得, 对任意 $p \ge 1$
		\[
			(a - \varepsilon)(y_n - y_{n+p}) < x_n - x_{n+p} < (a + \varepsilon)(y_n - y_{n+p}),
		\]
		令 $p \to +\infty$ 有
		\[
			y_n (a - \varepsilon) < x_n < y_n (a + \varepsilon),
		\]
		再同时除以 $y_n$ 有
		\[
			a - \varepsilon < \frac{x_n}{y_n} < a + \varepsilon.
		\]
		即
		\[
			\lim_{n\to\infty} \frac{x_n}{y_n} = a.
		\]
		\smallskip
		
		\item 当 $a = +\infty$ 时, 由无穷大量的定义可知, 对于给定 $C > 0$, 存在 $N \in \N$ 使得对任意 $n > N$ 有
		\[
			x_n − x_{n+1} > C (y_n − y_{n+1}),
		\]
		进一步有对任意 $n > N$ 和 $p \ge 1$ 有
		\[
			x_n − x_{n+p} > C (y_n − y_{n+p}).
		\]
		令 $p \to +\infty$ 得到
		\[
			x_n > C.
		\]
		即
		\[
			\lim_{n\to\infty} \frac{x_n}{y_n} = a.
		\]
		\smallskip
		
		\item 当 $a = -\infty$ 时, 可以用类似 $\frac{\infty}{\infty}$ 的证明方法证明.
	\ITEME
\end{proof}

\begin{remark}
注意两种情形下的区别, 前者为后项减前项, 后者为前项减后项.	
\end{remark}

\subsection{级数初步}
\begin{definition}{级数}
我们将表达式 $a_1 + a_2 + \cdots + a_n + \cdots$ 记作 
\[
	\sum\limits_{n=1}^\infty a_n \quad\text{或}\quad \sum\limits_{n \ge 1} a_n ,
\]
通常称为级数或无穷级数 (Infinite series). 其中数列 $\{a_n\}$ 的元素称为级数的项, 而 $a_n$ 称通项 (General form) 为级数的第 $n$ 项. 

\[
	s_n := \sum\limits_{k=1}^n a_k,
\] 
称为级数的部分和 (Partial sum) 或前 $n$ 项和.

如果级数的部分和的数列收敛就称级数收敛, 否则称级数发散. 如果级数部分和数列的极限存在, 则 $\lim\limits_{n\to\infty} s_n = s$ 称为级数的和 (Sum), 做出如下记号.
\[
	\sum_{n=1}^\infty a_n := s.
\]
\end{definition}

\begin{remark}
	收敛级数的本质是一类数列极限, 即部分和数列极限, 可以使用数列极限的性质.
\end{remark}

我们接下来简单分析一下收敛级数的性质.
\begin{theorem}{收敛级数的代数性质}
	设有两个收敛级数 $\sum\limits_{n = 1}^\infty a_n = a, \sum\limits_{n = 1}^\infty b_n = b$, 则对于任意的 $\alpha , \beta \in \R$, 数列 $\sum\limits_{n = 1}^\infty (\alpha a_n + \beta b_n)$ 也收敛, 且有
	\[
		\sum\limits_{n = 1}^\infty (\alpha a_n + \beta b_n) = \alpha\sum\limits_{n = 1}^\infty a_n + \beta\sum\limits_{n = 1}^\infty b_n.
	\]
\end{theorem}

\begin{proof}
	由\hyperref[thm:Slimit_p5]{数列极限代数性质}显然. 但要注意发散数列不一定满足上述性质.
\end{proof}

\begin{theorem}{收敛级数的必要条件}
	设有收敛级数 $\sum\limits_{n = 1}^\infty a_n = a$ 则
	\[
		\lim_{n \to \infty} a_n = 0.
	\]
\end{theorem}

\begin{proof}
	由 $a_n = s_n - s_{n-1}$ 易证.
\end{proof}

\section{数列极限的审敛法}
\subsection{Cauchy 准则}
\begin{theorem}{Cauchy 准则}{R5}
	对于数列 $\{x_n\}$ 收敛的充要条件是, 对于任意给定的正数 $\varepsilon$, 存在正整数 $N$, 使得当 $n, m > N$ 时, 满足 
	\[
		|x_n-x_m| < \varepsilon.
	\]
	我们称满足以上条件的数列为基本数列 (Cauchy 数列).
\end{theorem}

\begin{idea*}
必要性显然, 充分性通过构造上确界与下确界的闭区间套, 由闭区间套原理可证明.	
\end{idea*}

\begin{proof}
	我们首先证明必要性, 设 $\lim\limits_{n\to\infty} x_n = A$. 对于 $\forall \varepsilon > 0$, 可以求出序号 $N$, 使得当 $n > N$ 时有 $\lvert x_n - A\rvert < \varepsilon/2$. 如果现在 $m > N$ 且 $n > N$, 则 
	\[
		\lvert x_m - x_n \rvert = \lvert x_m - A - x_n + A \rvert \le \lvert x_m - A \rvert + \lvert x_n - A \rvert < \varepsilon/2 + \varepsilon/2 = \varepsilon,
	\]
	必要性证毕.
	
	接下来我们证明充分性, 设 $\{x_k\}$ 是一个基本数列. 对于 $\forall\varepsilon > 0$, 可以求出序号 $N$, 使得当 $m \ge N$ 且 $k\ge N$ 时有 $\lvert x_m - x_k \rvert < \varepsilon/3$. 把 $m$ 固定为 $m = N$, 则对于任何 $k > N$ 都有
	\[
		x_N - \frac\varepsilon3 < x_k < x_N +\frac\varepsilon3.
	\] 
	即 $\{x_k\}$ 是有界数列.
	
	我们做出如下定义 $a_n := \inf\limits_{k \ge n} x_k ,\ b_n := \sup\limits_{k \ge n} x_k$.
	
	则显然有 $a_n \le a_{n+1} \le b_{n+1} \le b_n$. 根据 \hyperref[thm:R2]{闭区间套原理} 可得闭区间套列 $[a_n, b_n]$ 存在一公共点 $A$. 
	
	对于任何 $n \in \N$我们有
	\[
		a_n \le A \le b_n,
	\]
	当 $k \ge n$ 时有
	\[
		a_n \le x_k \le b_n.
	\]	
	所以当 $k \ge n$ 时
	\[
		\lvert A - x_k \rvert \le b_n - a_n.
	\]
	综上可知
	\[
	x_N - \frac{\varepsilon}{3} \le a_n \le b_n \le x_N + \frac{\varepsilon}{3},
	\]
	所以有
	\[
		b_n - a_n \le x_N + \frac{\varepsilon}{3} - x_N + \frac{\varepsilon}{3} = \frac{2\varepsilon}{3} < \varepsilon.
	\]
	即
	\[
		\lvert A - x_k \rvert \le \varepsilon.
	\]
	从而 $\lim\limits_{n\to\infty} x_n = A$, 充分性得证.
	
\end{proof}
\smallskip

\subsection{单调数列极限存在准则}
\begin{definition}{单调性}
	对于数列 $\{x_n\}$, 如果 $\forall n\in\N\ (x_n < x_{n+1})$, 则称之为严格单调递增数列(Strictly increasing sequence); 如果 $\forall n\in\N\ (x_n \le x_{n+1})$, 则称之为单调递增数列 (Monotonically increasing sequence) 或单调不减数列；如果 $\forall n\in\N\ (x_n > x_{n+1})$, 则称之为严格单调递减数列 (Strictly decreasing sequence); 如果 $\forall n\in\N\ (x_n \ge x_{n+1})$, 则称之为单调递减数列 (Monotonically decreasing sequence) 或单调不增数列.
\end{definition}

% Weierstrass 定理
\begin{theorem}{单调数列极限存在准则 (Weierstrass 定理)}{R6}
单调不减数列的极限存在的充要条件是有上界.	
\end{theorem}

\begin{idea*}
	确界原理可证上确界为其极限.
\end{idea*}

\begin{proof}
	必要性由数列极限的有界性显然, 这里只用证明其充分性.
	
	由有界性可知 $\{x_n\}$ 存在上界, 由确界原理可知其有唯一的上确界 $s = \sup\limits_{n\in\N} x_n$.
	
	由上确界的定义可知, 对于任意的 $\varepsilon > 0$, $\exists x_N \in \{x_n\}$ 使得 $s - \varepsilon < x_N \le s$, 由于 $\{x_n\}$ 是不减数列, 所以对于任何 $n > N$ 都有 $s - x_n < \varepsilon$, 即有 $\lim\limits_{n \to \infty} x_n = s$. 可知极限等于其上确界.
\end{proof}
\smallskip

这里给出几个实例.

\begin{example}
	证明当 $q > 1$ 时, $\lim\limits_{n \to \infty} \frac{n}{q^n} = 0$. 
\end{example}

\begin{solution}
	由数列的通项公式可以求出数列的递推公式 $x_{n+1} = \frac{n+1}{nq} x_n$. 对其中系数求极限有
	\[
		\lim_{n \to \infty} \frac{n+1}{nq} = \lim_{n\to \infty} \left(1 + \frac{1}{n}\right)\frac{1}{q} = \lim_{n\to \infty} \left(1 + \frac{1}{n}\right) \cdot \lim_{n\to \infty} \frac{1}{q} = \frac{1}{q} < 0
	\]
	有极限定义可知存在 $N$ 使得对于 $\forall n > N$, 有 $\frac{n+1}{nq} < 1$. 即对于 $\forall n > N$ 时 $\{x_n\}$ 单调递减. 由于数列极限的有限项不影响数列的收敛性, 我们只需求出数列 $\{x_n\}_{n > N}$ 的极限.
	
	显然数列 $\{ x_n \}_{n > N}$ 的各项值恒大于 $0$, 即数列下有界, 根据 \hyperref[thm:R6]{Weierstrass 定理} 可知其极限存在.
	
	设 $x = \lim\limits_{n \to \infty} x_{n}$ 有
	\[
		x = \lim_{n \to \infty} x_{n + 1} = \lim_{n \to \infty}\left(\frac{n + 1}{nq} x_n\right) = \lim_{n \to \infty}\frac{n+1}{nq} \cdot \lim_{n \to \infty} x_n = \frac 1q x.
	\]
	则有 $(1 - \frac1q) x = 0$, 即 $x = \lim\limits_{n \to \infty} x_{n} = 0$.
\end{solution}
\smallskip

由上面的例题我们还可以得到如下的两个推论 
\[
	\lim_{n \to \infty} \sqrt[n]{n} = 1,
\]
\[
	\forall a > 0, \lim_{n \to \infty} \sqrt[n]{a} = 1.
\]
\smallskip

\begin{example}
	证明 $\lim\limits_{n \to \infty} \frac{q^n}{n!} = 0$, 其中 $q \in \R$, $n \in \N$.
\end{example}

\begin{solution}
	当 $q = 0$ 时结论显然成立. $q \neq 0$ 时因为 $\left\lvert \frac{q^n}{n!} \right\rvert = \frac{\lvert q \rvert ^n}{n!}$, 所以不妨设 $q > 0$.
	
	注意到该数列的通项公式为 $x_{n+1} = \frac{q}{n+1} x_n$, 考虑用 \hyperref[thm:R6]{Weierstrass 定理} 证明, 显然存在 $N > 0$ 使得对 $\forall n > N$ 都有 
	\[
		0 < \frac{q}{n+1} < 1
	\]
	即数列 $\{x_n\}_{n>N}$ 是严格单调递减数列, 而显然再由数列非负由 \hyperref[thm:R6]{Weierstrass 定理} 可得极限存在. 因此设极限为 $x$ 有
	\[
		x = \lim_{n\to\infty} x_{n+1} = \lim_{n\to\infty} \frac{q}{n+1} x_n = \lim_{n\to\infty} \frac{q}{n+1} \lim_{n\to\infty} x_n = 0 \cdot x = 0.
	\]
\end{solution}
\begin{remark}
	若极限不存在时不能设其为 $x$.
\end{remark}

\subsection{三个重要常数}

我们先来介绍我们最熟悉的常数, 圆周率 $\pi$, 即圆的周长与直径之比, 约等于 $3.14159$. 其符号取自 ``周长" 希腊语 ``πϵριµϵρoζ" 的首字母, 实际上 $\pi$ 在 1706 年才被 William Jones 在其著作 Synopsis Palmariorium Mathesios 中用来表示圆周率.

$\pi$ 之所以是这么重要的一个常数, 是因为 $\pi$ 与几何学的紧密联系, 无论是欧氏几何的圆, 周期现象还是旋转对称, 都会自然产生 $\pi$. 所以 $\pi$ 是几何内在的常数，不依赖几何上的尺度.

\smallskip

在十六世纪后, 我们常用无穷级数计算 $\pi$ 的值, 并由此衍生出了千奇百怪的公式, 它们之所以长得奇怪实际上是为了追求更快的收敛速度, 使每计算一项可以增加更多位的十进制精度, 例如 Ramanujan 著名的公式
\[
\frac{1}{\pi} = \frac{2\sqrt{2}}{9801}\sum_{n=0}^{\infty}\frac{(4n)!}{(n!)^4}\frac{1103+26390n}{396^{4n}}.
\]
以及受其启发, 现代计算 $\pi$ 常用的 Chudnovsky 公式, 计算机就使用此公式计算 $\pi$ 的值.
\[
\frac{1}{\pi} = 12\sum_{n=0}^{\infty}\frac{(-1)^n(6n)!(13591409+545140134n)}{(3n)!(n!)^3\,640320^{3n+3/2}}.
\]
\smallskip

与 $\pi$ 相关的还有一个有趣的问题, 著名的巴塞尔问题, 即计算所有自然数平方的导数和, 惊人的是全体自然数竟然以这种奇妙的方式与几何上的常数 $\pi$ 建立起了联系.
\begin{theorem}{巴塞尔问题}{basel}
	\[
		\sum_{n = 1}^{\infty} \frac{1}{n^{2}} = \frac{1}{1^{2}} + \frac{1}{2^{2}} + \frac{1}{3^{2}} + \frac{1}{4^{2}} + \cdots = \frac{\pi^{2}}{6}
	\]
\end{theorem}
巴塞尔问题由 Pietro Mengoli 于 1644 年提出. 1735 年, Euler 给出了第一个不严格的证明, 并于 1741 年给出严格的证明. 巴塞尔问题得名于 Euler 居住的瑞士城市巴塞尔. 由于与之相关的证明需使用积分与无穷级数的知识, 我将其放在 \hyperlink{A2}{附录 A.2}.
\bigskip

接下来我们介绍自然常数 $e$. Jacob Bernoulli 于 1683 年时研究连续复利问题时发现了常数 $e$. Euler在 1727 年开始用e来表示这个常数, 所以也常称 $e$ 为 Euler 数.

$e$ 在分析学中的重要性与算数中的 $1$ 和 几何中的 $\pi$ 的重要程度相当. 连续变化, 增长率, 指数过程都会自然产生 $e$, 它与指对幂函数与极限过程都有着深刻的联系.

我们先介绍连续复利问题, 连续复利是指利息在无限短的间隔内进行计算利息并加入本金产生新利息，即计息频率趋于无穷大的极限模型. 其中我们假设利息为计息频率的倒数, 则有下式
\[
	\lim_{n\to\infty} \left(1 + \frac{1}{n}\right)^n.
\]

接下来我们证明该数列收敛. 首先我们给出以下引理.
% Bernoulli 不等式
\begin{lemma}{Bernoulli 不等式}{bernoulli}
	对于 $n \in \N, \alpha > -1$ 有
	\[
		(1 + \alpha)^n \ge 1 + n\alpha.
	\]	
\end{lemma}
\begin{proof}
	$n = 1$ 时结论显然成立, 考虑用数学归纳法证明. 假设 $(1 + \alpha)^n \ge 1 + n\alpha$ 成立, 证明 $(1 + \alpha)^{n+1} \ge 1 + (n + 1)\alpha$ 成立.
	\[
		(1 + \alpha)^{n+1} = (1 + \alpha) (1 + \alpha)^n \ge (1 + \alpha) (1 + n\alpha) = 1 + \alpha + n\alpha + n \alpha^2 \ge 1 + (n + 1)\alpha.
	\]
	于是由归纳原理可知 Bernoulli 不等式成立.
\end{proof}

\begin{example}
	证明数列 $a_n = \left\{ \left(1 + \frac{1}{n}\right)^n \right\}$ 收敛.
\end{example}

\begin{solution}
	构造如下数列 
	\[
		b_n = a_n \cdot \left(1 + \frac{1}{n}\right) = \left(1 + \frac{1}{n}\right)^{n+1},
	\]
	判断 $b_n$ 的单调性, 即比较 $\frac{b_{n-1}}{b_n}$ 与 $1$ 的大小.
	\[
		\frac{b_{n-1}}{b_n} = \frac{\left(1 + \frac{1}{n - 1}\right)^{n}}{\left(1 + \frac{1}{n}\right)^{n+1}} = \frac{\left(\frac{n}{n - 1}\right)^{n}}{\left(\frac{n + 1}{n}\right)^{n+1}} = \frac{n^{2n}}{(n^2 -1)^n} \cdot \frac{n}{n + 1} = \left(1 +\frac{1}{n^2 -1}\right)^n \cdot \frac{n}{n + 1},
	\]
	利用 \hyperref[lem:bernoulli]{Bernoulli 不等式} 可得
	\[
		\frac{b_{n-1}}{b_n} = \left(1 +\frac{1}{n^2 -1}\right)^n \cdot \frac{n}{n + 1} \ge \left(1 + \frac{n}{n^2 -1}\right) \cdot \frac{n}{n + 1} > \left(1 + \frac{1}{n}\right) \cdot \frac{n}{n + 1} = 1.
	\]
	则可知 $b_n$ 单调递减, 显然 $b_n$ 是非负数列, 所以由 \hyperref[thm:R6]{Weierstrass 定理} 有 $\lim\limits_{n\to\infty}\left\{ \left(1 + \frac{1}{n}\right)^{n+1} \right\}$ 收敛. 进一步可得
	\[
		\lim\limits_{n\to\infty} a_n = \lim_{n\to\infty} b_n \left(1 + \frac{1}{n}\right)^{-1} = \lim_{n\to\infty} b_n \cdot \lim_{n\to\infty}\left(1 + \frac{1}{n}\right)^{-1} = \lim_{n\to\infty} b_n.
	\]
	所以数列 $\left\{ \left(1 + \frac{1}{n}\right)^n \right\}$ 收敛
\end{solution}

由此我们有如下定义.

\begin{definition}{自然常数}{e}
\[
	e := \lim_{n\to\infty} \left(1 + \frac{1}{n}\right)^n.
\]
\end{definition}

根据牛顿二项式定理公式,
\[
	\left(1 + \frac1n\right)^n = \sum_{i = 0}^{n} \frac{1}{i!} \frac{n!}{(n-i)!n^i},
\]
取
\[
	e_n =  \left(1 + \frac1n\right)^n,\ s_n = \sum_{i = 0}^{n} \frac{1}{i!},
\]
显然有 $e_n < s_n$.

从以上展开式可见, 对于任何固定的 $k$, 当 $n \ge k$ 时有
\[
	\sum_{i = 0}^{k} \frac{1}{i!} \frac{n!}{(n-i)!n^i} \le e_n = \sum_{i = 0}^{n} \frac{1}{i!} \frac{n!}{(n-i)!n^i}.
\]
等式左边有
\[
	\sum_{i = 0}^{k} \frac{1}{i!} \frac{n!}{(n-i)!n^i} = 1 + 1 + \frac{1}{2!}\left(1-\frac1n\right) + \cdots + \frac{1}{k!}\left(1-\frac1n\right)\cdots\left(1-\frac{k-1}{n}\right).
\]
令 $n \to \infty$ , 
\[
	\lim_{n\to\infty}\sum_{i = 0}^{k} \frac{1}{i!} \frac{n!}{(n-i)!n^i} = 1 + 1 + \frac{1}{2!} + \cdots + \frac{1}{k!}
	= s_k.
\]
可见等式左边趋于 $s_k$, 而右边显然趋于 $e$, 所以对于任何 $k \in \N$ 都有 $s_k \le e$. 得到如下关系式
\[
	e_n < s_n \le e.
\]
这时再令 $n \to \infty$, 根据迫敛性可得 $\lim\limits_{n\to\infty} s_n = e$.

根据级数的和的定义, 我们现在可以写出
\[
	e = 1 + \frac1{1!} + \frac1{2!} + \cdots + \frac1{n!} + \cdots = \sum_{n=0}^\infty \frac{1}{n!}.
\]

级数形式是 $e$ 的一个特别便于计算的表达式.

除此之外对 $e$ 还有以下定理.

\]	
\end{theorem}

\begin{proof}

\end{proof}

最后介绍一个十分神秘的数学常数 $\gamma$, 我们称之为 Euler 常数. 实际上我们根本不知道 $\gamma$ 是有理数还是无理数

前面我们给出了著名的 \hyperref[thm:basel]{巴塞尔问题}, 即所有正整数的二次幂的倒数和. 现在考虑一般情况, 即所有正整数的 $p > 0$ 次幂的倒数和, 即如下级数
\[
	\sum_{n = 1}^{\infty} \frac{1}{n^p}.
\]
有部分和序列
\[
	s_n = \sum_{k = 1}^{k} \frac{1}{k^p}.
\]

若把 $p$ 换成复数幂就得到了著名的 Riemann $\zeta$ 函数.

\begin{definition}{Riemann $\zeta$ 函数}
	设有复数 $s \in \C$, 写为 $s = \sigma + it$. 当实部 $\sigma > 1$ 时, 定义为
	\[
		\zeta(z) := \sum_{n = 1}^{\infty} \frac{1}{n^s}.
	\]
\end{definition}

我们回到 $p \in \R^+$ 的情况, 首先我们注意到其部分和数列显然是单调递增的, 所以只要我们能证明其上有界则可证明级数收敛. 

首先我们将部分和数列按 $2$ 的幂分组,
\[
s_{2^m} = 1 + \left(\frac1{2^p}\right) + \left(\frac1{3^p}+\frac1{4^p}\right) + \left(\frac1{5^p}+\cdots+\frac1{8^p}\right) + \cdots + \left(\frac1{(2^{m-1}+1)^p} + \cdots+\frac1{(2^m)^p}\right).
\]
对于第 $k$ 组, 其包含 $2^{k-1}$ 个项, 且有对于组内任意项有
\[
	\frac1{n^p} \le \frac1{(2^{k-1})^p},
\]
则可得
\[
	\sum_{n = 2^{k-1}+1}^{2^k} \frac1{n^p} \le \frac{2^{k-1}}{2^{(k-1)p}} = 2^{(k-1)(1-p)}.
\]
进一步有
\[
	s_{2^m} = 1 + \sum_{k=1}^{m}\sum_{n = 2^{k-1}+1}^{2^k} \frac1{n^p} \le 1 + \sum_{k=1}^{m} 2^{(k-1)(1-p)} = 1 + \sum_{i = 0}^{m-1} \left(2^{(1-p)}\right)^{i}.
\]
注意到 $\sum\limits_{i = 0}^{m-1} \left(2^{1-p}\right)^{i}$ 是一个几何级数, 则 $2^{1-p} < 1$ 时级数收敛, 即 $p > 1$ 时级数收敛. 类似的, 对于 $\zeta(z)$ 当且仅当 $\Re (z) > 1$ 时收敛.
  
当级数 $\sum\limits_{n = 1}^{\infty} \frac{1}{n^p}$ 中 $p = 1$ 时, 我们称之为调和数列 (Harmonic series), 之所以称为调和级数, 是因为每一项都是其前后两项的调和中项 (音律中，如果一个弦的长度是另一个弦的倒数比例, 它们会发出和谐的声音). 现在我们考虑如下极限.
\[
	\lim_{n\to\infty} \frac{\sum\limits_{n = 1}^{\infty} \frac{1}{n}}{\ln n}
\]
显然这是一个 $\frac{\infty}{\infty}$ 型极限, 我们可以对其应用 \hyperref[thm:stolz_1]{Stolz 定理}, 则有
\[
	\begin{aligned}
		\lim_{n\to\infty} \frac{\sum\limits_{n = 1}^{\infty} \frac{1}{n}}{\ln n} &= \lim_{n\to\infty} \frac{\frac{1}{n}}{\ln n - \ln (n - 1)} = \lim_{n\to\infty} \frac{\frac{1}{n}}{\ln \frac{n}{n - 1}} \\
		&= \lim_{n\to\infty} \frac{\frac{1}{n}}{\ln \left(1 + \frac{1}{n}\right)} \cdot \lim_{n\to\infty} \frac{\ln \left(1 + \frac{1}{n}\right)}{\ln \left(1+\frac{1}{n - 1}\right)}\\
		&= \lim_{n\to\infty} \frac{1}{\ln \left(1 + \frac{1}{n}\right)^n} \cdot \frac{\frac{1}{n}}{\frac{1}{n - 1}}\\
		&= 1.
	\end{aligned}
\]
即有
\[
	\sum_{n = 1}^{\infty}\frac{1}{n} \sim \ln n \quad (n \to \infty).
\]
由此我们定义如下数列
\[
	a_n := \sum_{k = 1}^{n}\frac{1}{k} - \ln n.
\]
则有
\[
	a_{n+1} - a_n = \frac{1}{n+1} - \ln (n+1) + \ln n = \frac{1}{n+1} - \ln \left(1+\frac1n\right) < 0.
\]
即
\[
	a_n > a_{n+1} > 0,
\]
根据 \hyperref[thm:R6]{Weierstrass 定理}, $\lim\limits_{n\to\infty}a_n$ 存在

\begin{definition}{Euler 常数}
	Euler 常数 $\gamma$ 定义为
	\[
		\gamma := \lim_{b\to\infty}\left(\sum_{k=1}^n\frac{1}{k} − \ln n \right) = 0.577215664901532\cdots.
	\]
\end{definition}

Alexander Aptekarev 在 2009 年证明了 Euler 常数 $\gamma$ 和 Euler-Gompertz 常数 $\delta$ 至少有一个是无理的, 则我们有以下猜想.

\begin{conjecture}
$\gamma$ 是无理数.	
\end{conjecture}

\subsection{数列与部分极限}
% 子列
\begin{definition}{子列}
如果 $x_1, x_2, \cdots, x_n, \cdots$ 是某数列, 而 $n_1 < n_2 < \cdots < n_k < \cdots$ 是严格单调递增自然数列, 则数列 $x_{n_1}, x_{n_2}, \cdots, x_{n_k}, \cdots$ 称为数列的子列.
\end{definition}

我们曾经介绍过 Bolzano-Weierstrass 原理 (\hyperref[thm:R4]{极限点定理}) , 在这里给出致密性定理, 即其另一等价表述.
\begin{theorem}{Bolzano-Weierstrass 原理}{R4'}
	每个有界实数列都含有收敛的子列.
\end{theorem}
\begin{proof}
	设 $E$ 是有界数列 $\{x_n\}$ 的值集, 如果 $E$ 是有限集, 则显然存在收敛子列. 若 $E$ 是无限集, 则根据 \hyperref[thm:R4]{极限点定理} 可知 $E$ 至少存在一个极限点 $x$. 由极限点定义可知存在子列收敛于 $x$.
\end{proof}

\begin{corollary}
	从每一个实数列中都可以选出一个收敛的或趋于无穷的子列.
\end{corollary}

\begin{proof}
	设有无界数列 $\{x_n\}$ , 则存在数列 $\{n_k\}$ 其中 $n_k \in \N$ 使得 $\lvert x_{n_k} \rvert > k$ 且 $n_k < n_{k+1}$, 则 $\{x_{n_k}\}$ 趋于无穷. 
\end{proof}


% 上下极限
\begin{definition}{上下极限}
设有数列 $\{x_n\}$, 我们称对于任意 $k$ 数列 $\{x_n\}_{n \ge k}$ 的上确界 (下确界) 的极限为上极限 (下极限) 记为
\[
	\varlimsup_{n\to\infty} x_n \equiv \limsup_{n\to\infty} x_n := \lim_{k\to\infty} \sup_{k \ge n} x_k,
\]
\[
	\varliminf_{n\to\infty} x_n \equiv \liminf_{n\to\infty} x_n := \lim_{k\to\infty} \inf_{k \ge n} x_k.
\]
\end{definition}

\begin{proposition}
	任何数列的上极限与下极限分别为部分极限中的最大值和最小值.
\end{proposition}

\begin{proof}
	先考虑数列有界的情况, 我们以下极限 $i = \varliminf_{k\to\infty} x_k$ 为例, 则有数列 $i_n = \inf_{k \ge n} x_k$ 是不减数列. 对于 $n \in \N^+$, 根据下确界的定义, 按照归纳原理可以求出 $k_n \in \N^+$, 使得 $k_n - k_{n+1}$ 且
	\[
		i_{k_{n-1} + 1} \le x_{k_n} < i_{k_{n-1} + 1} + \frac{1}{n},\ (k_0 = 0)
	\]
	令 $n \to \infty$ 可得
	\[
		\lim_{n \to \infty} i_{k_{n-1} + 1} = i,
	\]
	\[
		\lim_{n \to \infty} \left(i_{k_{n-1} + 1} + \frac{1}{n}\right) = i,
	\]
	根据数列极限的 \hyperref[thm:Ssandwich]{迫敛性} 可得
	\[
		\lim_{n \to \infty} x_{k_n} = i.
	\]
	则 $i$ 是一个极限点, 因为对于 $\forall \varepsilon > 0$, 可以求出 $n \in \N$ 使得 $i - \varepsilon < i_n$ 所以任何极限点都不能小于 $i$. 类似的可以证明上确界是极限点的最大值.
	无界数列显然.
\end{proof}

\begin{corollary}{}{liminfeqlimsup}
	数列有极限, 或趋于无穷的充要条件是上下极限相等.
\end{corollary}

\begin{proof}
	数列极限的 \hyperref[thm:Ssandwich]{迫敛性} 易证.
\end{proof}

\begin{corollary}
	数列收敛的充要条件是数列的子列也都收敛.
\end{corollary}

\begin{proof}
	由推论 \ref{cor:liminfeqlimsup} 可知数列极限存在时其上下极限重合, 子列的上下极限介于原数列上下极限之间, 则子列的上下极限也重合, 则其子列与原数列收敛于同一极限.
\end{proof}

\section{函数极限}

\subsection{函数极限的定义}

首先给出在无穷远处的函数极限.

\begin{definition}{无穷远处的函数极限}
	\ITEMS
		\item 假设函数为 $f : (a,+\infty) \rightarrow \R$ 和实数 $A \in \R$, 若对于 $\forall \varepsilon > 0$, $\exists M > a$, $\forall x > M$ 都有
		\[
			\lvert f(x) - A \rvert < \varepsilon.
		\]
		则 $\lim\limits_{x\to+\infty} f(x) = A$.
		\bigskip
		
		\item 假设函数为 $f : (-\infty,b) \rightarrow \R$ 和实数 $A \in \R$, 若对于 $\forall \varepsilon > 0$, $\exists M < b$, $\forall x < M$ 都有
		\[
			\lvert f(x) - A \rvert < \varepsilon.
		\]
		则 $\lim\limits_{x\to-\infty} f(x) = A$.
		\bigskip
		
		\item 假设函数为 $f : (-\infty,+\infty) \rightarrow \R$ 和实数 $A \in \R$, 若对于 $\forall \varepsilon > 0$, $\exists M > 0$, $\forall \lvert x\rvert > M$ 都有
		\[
			\lvert f(x) - A \rvert < \varepsilon.
		\]
		则 $\lim\limits_{x\to\infty} f(x) = A$.
	
	\ITEME
\end{definition}

% 空心邻域
\begin{definition}{空心邻域}
从一个点的邻域去掉该点本身, 所得集合为该点的空心邻域或去心邻域. 我们引入如下记号
\[
	\mathring{U}(a,\rho) := (a - \rho, a + \rho)\setminus \{a\}= (a - \rho,a) \cup (a,a + \rho).
\]
称为点 $a$ 的去心邻域.
\end{definition}

我们现给出函数极限的第一种定义方式, 也就是最早由 Cauchy 提出的定义, 即 $\varepsilon$-$\delta$ 语言.

\begin{definition}{$\varepsilon$-$\delta$ 语言}{e-d}
对于函数 $f:\mathring{U}(a,\rho) \rightarrow \R$, 和 $A\in\R$ 其中 $f(a) = A$. 如果对于任何 $\varepsilon > 0$, 都存在 $\delta > 0$ 使得对于满足 
\[
	\lvert x - a \rvert < \delta,
\] 
的任意 $x \in E$ 都有
\[
	\lvert f(x) - A \rvert < \varepsilon,
\] 
我们说, 当 $x$ 趋于 $a$ 时, 函数 $f$ 趋于 $A$, 或者说 $A$ 是 $f$ 在 $x$ 趋于 $a$ 时的极限. 记为
\[
	\lim_{x\to\infty} f(x) = A.
\]

我们也可以用形式符号描述上述定义
\[
	\forall \varepsilon > 0 \ \exists \delta > 0\ \forall a \in E\ (0 < \lvert x - a \rvert < \delta \Rightarrow \lvert f(x) - A \rvert < \varepsilon).
\]
\end{definition}

我们可以给出函数极限的等价表述.
\begin{definition}{函数极限邻域定义}
如果对于点 $A$ 任何一个邻域, 都可以找到点 $a$ 在集合 $E$ 中的一个去心邻域, 使得它在映射 $f: E \to \R$ 下的像包含于点 $A$ 的给定邻域, 则数 $A$ 称为函数 $f: E \to \R$ 在 $x$ 沿集合 $E$ 趋于点 $a$ 时的极限.

用形式符号上述定义为
\[
	\left( \lim_{E \ni x \to a} f(x) = A\right) := \forall V_{\R}(A)\ \exists \mathring{U}_E(a) \left( f\left( \mathring{U}_E(a)\right) \subset V_{\R}(A)\right).
\]
\end{definition}

\begin{definition}{单侧极限}{onesidelimit}
\ITEMS
	\item 对于函数 $f: (a-\rho,a) \rightarrow \R$, 和 $A\in\R$ 其中 $f(a) = A$.如果对于任何 $\varepsilon > 0$, 都存在 $\delta \in (0,\rho]$ 使得对于任意 $x \in (a-\delta,a)$ 都有
	\[
		\lvert f(x) - A \rvert < \varepsilon,
	\] 
	则称 $A$ 为 $f(x)$ 在 $a$ 的左极限 (Left-hand limit), 记为
	\[
		f(a-) := \lim_{x\to\infty-} f(x) = A.
	\]
	\smallskip

	\item 对于函数 $f: (a,a+\rho) \rightarrow \R$, 和 $A\in\R$ 其中 $f(a) = A$.如果对于任何 $\varepsilon > 0$, 都存在 $\delta \in (0,\rho]$ 使得对于任意 $x \in (a,a+\delta)$ 都有
	\[
		\lvert f(x) - A \rvert < \varepsilon,
	\] 
	则称 $A$ 为 $f(x)$ 在 $a$ 的左极限 (Left-hand limit), 记为
	\[
		f(a+) := \lim_{x\to\infty+} f(x) = A.
	\]

\ITEME
\end{definition}


\begin{theorem}
	函数 $f$ 在点 $a$ 收敛且极限为 $A$ 的充要条件是 $f$ 在 $a$ 点的左右极限存在且等于 $A$, 即
	\[
	\lim_{x\to a} f(x) = A\quad \iff\quad f(a-) = A = f(a+).
	\]
\end{theorem}

\begin{proof}
	由定义显然.
\end{proof}


\begin{example}
	符号函数
	\[
		\mathrm{sgn} (x) = \begin{cases}
			1, & x>0,\\
			0, & x=0,\\
			-1, & x<0
		\end{cases}
	\]
	是在整个数轴上定义的. 证明在 $x$ 趋于 $0$ 时，它没有极限.
\end{example}

\begin{solution}
	显然 $\mathrm{sgn} (0-) = -1$, $\mathrm{sgn} (0+) = 1$ 左右极限不相等, 极限不存在.
\end{solution}

\begin{definition}{无穷极限}
对于函数 $f:\mathring{U}(a,\rho) \rightarrow \R$, 如果对于任何 $E > 0$, 都存在 $\delta > 0$ 使得对于满足 
\[
	\lvert x - a \rvert < \delta,
\] 
的任意 $x \in E$ 都有
\[
	\lvert f(x) \rvert > E.
\] 
称极限不存在, 记 $\lim\limits_{x\to a} f(x) = \infty$.
\end{definition}


\subsection{函数极限的性质}

% 唯一性
\begin{theorem}{函数极限的唯一性}{Flimit_p1}
函数在同一个点不可能有两个不同的极限.
\end{theorem}
\begin{idea*}
使用反证法, 假设函数在一点有两个不同的极限, 证明两个极限相同.
\end{idea*}

\begin{proof}
类比数列极限易证.
\end{proof}
\smallskip

% 有界性
\begin{theorem}{函数极限的局部有界性}{Flimit_p2}
若函数在某一点 $a$ 处有极限, 则函数在 $a$ 的某个去心邻域必有界.
\end{theorem}

\begin{idea*}
显然由定义可知, 存在 $\delta$ 使得当 $x$ 在 $a$ 的 $\delta$ 空心邻域上, 那么 $f(x)$ 将在 $A$ 的空心邻域上.
\end{idea*}

\begin{proof}
类比数列极限易证.
\end{proof}
\smallskip

% 保序性
\begin{theorem}{函数极限的局部保序性}{Flimit_p3}
设有 $\lim\limits_{x \to a} f(x) = A, \lim\limits_{x \to a} g(x) = B$, 设有 $A > B$ 则存在 $a$ 的某个空心邻域 $\mathring{U}(a,\rho)$ 上有
\[
	f(x) > g(x).
\]
\end{theorem}

\begin{idea*}
构建介于 $A, B$ 之间一点, 则存在一个 $x$ 的邻域使得两个极限的邻域都不包含该点.
\end{idea*}

\begin{proof}
类比数列极限易证.
\end{proof}
\smallskip

% 绝对值
\begin{theorem}{函数极限的绝对值性质}{Flimit_p4}
设有 $\lim\limits_{x \to a} f(x) = A$ 则有
\[
	\lim\limits_{x \to a} \lvert f(x) \rvert = \lvert A \rvert.
\]
\end{theorem}

\begin{idea*}
应用反向三角不等式可证.
\end{idea*}

\begin{proof}
根据极限的定义, 对于任意的 $\varepsilon > 0$, 存在 $\delta > 0$ 使得对于任意 $x \in \mathring{U}(a,\delta)$ 都有
\[
	\lvert f(x) - A \rvert < \varepsilon.
\]
则有
\[
	\bigl\lvert \lvert f(x) \rvert - \lvert A \rvert \bigr\rvert \le \lvert f(x) - A \rvert < \varepsilon.
\]
即
\[
	\lim_{x \to a} \lvert f(x) \rvert = \lvert A \rvert.
\]
\end{proof}
\smallskip

% 迫敛性
\begin{theorem}{函数极限的迫敛性}{Flimit_p5}
	设有函数 $f(x) , g(x), h(x)$, 若 $x$ 在某空心邻域上有 $g(x) \le f(x) \le h(x)$, 且有 $\lim\limits_{x \to a} g(x) = \lim\limits_{x \to a} h(x) = A$. 则有
	\[
		\lim_{x \to a} f(x) = A.
	\]
\end{theorem}

\begin{idea*}
由定义易证, 取更小的 $\delta$ 邻域即可.
\end{idea*}

\begin{proof}
类比数列极限易证.
\end{proof}
\smallskip

% 四则运算
\begin{theorem}{函数极限的四则运算}{Flimit_p6}
设 $f(x), g(x)$ 是两个函数, $\alpha, \beta$ 是两个常数, 如果 $\lim\limits_{x\to x_0} f(x) = A, \ \lim\limits_{x\to x_0} g(x) = B$, 则
	\ITEMS
		\item $\lim\limits_{x\to x_0} (\alpha f(x) + \beta g(x)) = \alpha A + \beta B$.
		\item $\lim\limits_{x\to x_0} f(x) \cdot g(x) = A \cdot B$.
		\item 当 $y_n\neq 0, \ n\in\N$ 且 $B \neq 0$ 时 $\lim\limits_{x\to x_0} \frac{f(x)}{g(x)} = \frac AB$.
	\ITEME
\end{theorem}

\begin{idea*}
类比 \hyperref[thm:Slimit_p5]{数列极限的四则运算}.
\end{idea*}

\begin{proof}
类比数列极限易证.
\end{proof}
\smallskip

% 复合函数极限
\begin{theorem}{复合函数极限}{Flimit_p7}
设有函数 $f(u), g(x)$, 如果 $\lim\limits_{u\to u_0}f(u) = A, \ \lim\limits_{x\to x_0}g(x) = u_0$ 且在 $x_0$ 的某空心邻域上有 $g(x) \neq u_0$ 则复合函数 $f \circ g$ 在 $x_0$ 处极限存在且
\[
	\lim_{x\to x_0}f[g(x)] = A.
\]
\end{theorem}

\begin{idea*}
根据函数极限定义, 从 $x_0$ 领域先对应到 $u_0$ 的领域再对应到 $A$ 的邻域.
\end{idea*}

\begin{proof}
根据极限的定义对于任意 $\varepsilon > 0$, 则存在 $\eta > 0$ 使得对于任意 $u \in \mathring{U}(u_0,\eta)$ 都有
\[
	\lvert f(u) - A \rvert < \varepsilon.
\]
再根据极限定义, 对这个 $eta$ 存在 $\delta$ 使得对于任意 $x \in \mathring{U}(x_0,\delta)$ 都有
\[
	\lvert g(x) - u_0 \rvert < \eta.
\]
即对于 $x \in \mathring{U}(x_0,\delta)$ 有
\[
	\lvert f[g(x)] - A \rvert < \varepsilon.
\]

\end{proof}
\smallskip

接下来我们介绍第二个重要极限, 即以函数极限得到的 $e$. 我们先引入如下取整符号.

\begin{definition}{取整符号}
	对于任意实数 $x$, 记 $\lfloor x \rfloor = n$ (也称为 Gauss 符号 $[x]$), 其中 $n$ 是唯一的整数, 且满足 $n \le x < n + 1$; $\lceil x \rceil = n$, 其中 $n$ 是唯一的整数, 且满足 $n - 1 < x \le n$.
\end{definition}

\begin{theorem}
\[
	\lim_{x \to \infty} \left(1 + \frac{1}{x}\right)^x = e.
\]
\end{theorem}
\begin{idea*}
	由 $e$ 的数列极限定义可推出如上函数极限, 利用迫敛性构造函数, 用取整符号使得构造出的函数可以应用数列极限. 
\end{idea*}
\begin{proof}
	构造如下不等式, 
	\[
		\left(1 + \frac{1}{\lfloor x \rfloor + 1}\right)^{\lfloor x \rfloor} < \left(1 + \frac{1}{x}\right)^{x} < \left(1 + \frac{1}{\lfloor x \rfloor}\right)^{\lfloor x \rfloor + 1},
	\]
	由 \hyperref[def:e]{$e$ 的定义} 可知
	\[
		\lim_{n \to \infty} (1 + \frac{1}{n + 1})^n = \lim_{n \to \infty} (1 + \frac{1}{n})^{n+1} = e.
	\]
	由迫敛性可知
	\[
		\lim_{x \to +\infty} (1 + \frac{1}{x})^x = e.
	\]
	在考虑 $x \to -\infty$ 的情况, 令 $y = -x$ 则有
	\[
	\begin{aligned}
		\lim_{x \to -\infty} \left(1 + \frac{1}{x}\right)^x &= \lim_{y \to +\infty} \left(1 - \frac{1}{y}\right)^{-y} = \lim_{y \to +\infty} \left(\frac{y - 1}{y}\right)^{-y} = \lim_{y \to +\infty} \left(\frac{y}{y - 1}\right)^{y}\\
		&= \lim_{y \to +\infty} \left[\left(1 + \frac{1}{y - 1}\right)^{y - 1}\left(1 + \frac{1}{y - 1}\right)\right] = e.
	\end{aligned}
	\]
	综上可得
	\[
		\lim_{x \to \infty} \left(1 + \frac{1}{x}\right)^x = e.
	\]
\end{proof}

\subsection{连续性的基本定义}

\begin{definition}{连续性}
假设函数 $f$ 在点 $a$ 的整个邻域上有定义, 我们说函数 $f$ 在点 $a$ 连续, 如果对于函数在点 $a$ 的值 $f(x)$ 的任何一个邻域 $V(f(a))$, 都可以找到点 $a$ 的邻域 $U(a)$, 使它在映射 $f$ 下的像包含于 $V(f(a))$.

这里给出连续性的形式逻辑写法, 同时给出常用的两个变体
\[
	\forall V(f(a))\ \exists U(a)\ (f(U(a)) \subset V(f(a))),
\]
\[
	\forall \varepsilon > 0\ \exists U(a)\ \forall x \in U(a)\ (\lvert f(x) - f(a)\rvert < \varepsilon),
\]
\[
	\forall \varepsilon > 0\ \exists \delta > 0\ \forall x\in \R\ (\lvert x - a\rvert < \delta \Rightarrow \lvert f(x) - f(a)\rvert < \varepsilon).
\]
\end{definition}

接下来我们考虑更一般的情况, 设函数 $f: E \to \R$ 定义在某个集合上的实值函数, 而 $a$ 是函数定义域中的一个点.

我们说函数 $f: E \to \R$ 在点 $a \in E$ 连续, 如果对于函数在点 $a$ 的值 $f(a)$ 的任何一个邻域 $V(f(a))$, 可以找到点 $a$ 在集合 $E$ 中的一个邻域 $U_E(a)$, 使得它的像 $f(U_E(a))$ 包含于 $V(f(a))$.

于是,
\[
	\forall V (f(a))\ \exists U_E (a)\ (f(U_E(a)) \subset V(f(a)))
\]

\begin{remark}
在提出连续性的严谨定义以前，Leibniz 曾定义连续函数就是图像可以一笔画出来, 不需要抬笔的函数, Euler 定义连续函数是用单一函数表示的一条曲线.
\end{remark}

\subsection{间断点}

如果函数 $f: E \to \R$ 在集合 $E$ 的某个点上不连续, 则该点称为函数 $f$ 的间断点.



点 $a\in E$ 称为函数 $f:E \to \R$ 的第一类间断点, 如果如果极限
\[
\begin{aligned}
	&\lim_{E\ni x \to a - 0} f(x),\\
	&\lim_{E\ni x \to a + 0} f(x)
\end{aligned}
\]
都存在, 但至少其中一个极限不等于函数在点 $a$ 的值 $f(a)$

如果点 $a\in E$ 是函数 $f: E \to \R$ 的间断点, 并且满足 $f|_{E\setminus a} = \widetilde{f}|_{E\setminus a}$ 的连续函数 $\widetilde{f}: E \to \R$, 则 $a$ 称为函数 $f: E \to \R$ 的可去间断点.

\subsection{向量及其表示}

\begin{definition}{向量与数量}
	我们称只有大小的量为数量(标量), 把既有大小又有方向的量称为向量(矢量).
\end{definition}

我们常常使用有向线段来表示向量, 有向线段方向代表向量的方向, 有向线段长度代表向量大小. 记为 $\Ve{AB}$, 其中 $A$ 是向量的起点, $B$ 是向量的终点. 有时我们也用黑体小写字母 $\ve a$ 表示向量(手写时仍用箭头表示).

\begin{definition}{模}
	向量 $\Ve{AB}$ 的大小, 也称为向量的模, 记作 $\lvert \Ve{AB} \rvert$.
\end{definition}

\begin{definition}{零向量}
	长度为零的向量称为零向量, 记作 $\ve{0}$.	
\end{definition}

\begin{definition}{共线}
	对于一组向量, 若将其平移至同一起点后, 它们在同一直线上, 那么就称这组向量是共线的(平行的), 用记号 $\parallelsum$ 表示.	
\end{definition}

零向量的的方向不定, 零向量与任何向量共线.

\begin{definition}{共面}
	对于一组向量, 若将其平移至同一起点后向量在同一平面上, 那么就称这组向量是共面的.
\end{definition}

显然, 任意两个共线的必共面.

\begin{definition}{共线}{collinearity}
	两个向量 $\ve a,\ve b$ 称为共线 (Collinearity), 或称平行, 若它们线性相关, 即存在不全为零的 $\alpha,\beta$, 使得

    \]
\end{definition}

\subsection{向量的内积}

\begin{definition}{内积}
	两个向量 $\ve a, \ve b$ 的内积 (Inner product) 是一个数量, 记为 $\ve a \cdot \ve b$, 所以也称为数量积或点乘. 其等于这两个向量的模乘以两向量夹角的余弦, 即
	\[
		\ve a\cdot\ve b := \lvert \ve a \rvert \lvert \ve b \rvert \cos \langle\ve a,\ve b\rangle.
	\]
	若用坐标表示两个向量则有
	\[
	\ve a\cdot\ve b = a_1b_1+\cdots+a_nb_n=\sum_{i=1}^n a_i b_i.
	\]
\end{definition}

显然两向量垂直时, 即 $\langle\ve a,\ve b\rangle = \frac{\pi}{2}$ 时有如下定理.

\begin{theorem}{内积与垂直}{}
	两个非零向量垂直的充要条件是其内积为零, 即
	\[
		\ve a \perp \ve b \quad \iff \quad \ve a\cdot\ve b =0.
	\]
\end{theorem}

除此之外, 利用内积可以用于计算两向量的夹角,
\[
\cos \langle \ve a, \ve b \rangle = \frac{\ve a \cdot \ve b}{\lvert \ve a \rvert \lvert \ve b \rvert}.
\]

内积也常常用来计算向量的长度, 对于任意向量 $\ve a$ 都有
\[
	\ve a ^2 := \ve a \cdot \ve a = \lvert \ve a \rvert \lvert \ve a \rvert \cos 0 = \lvert \ve a \rvert ^2,
\]
于是有
\[
\lvert a \rvert = \sqrt{\ve a ^2}.
\]

内积也可以用投影表示, 或者用内积表示投影, 如
\[
	\ve a \cdot \ve b = \lvert \ve b \rvert \rm{Prj}_{\ve b} \ve a (= \lvert \ve a \rvert \rm{Prj}_{\ve a} \ve b),
\]
\[
	\rm{Prj}_{\ve b} \ve a  = \frac{\ve a \cdot \ve b}{\lvert \ve b \rvert}.
\]

\subsection{内积的性质}

\begin{theorem}{内积运算律} 
对于任意向量 $\ve a, \ve b, \ve c$ 及任意实数 $\lambda$, 有
\ITEMS
	\item 交换律
	\[
		\ve a \cdot \ve b = \ve b \cdot \ve a.
	\]
	\item 与数乘的结合律
	\[
		(\lambda \ve a) \cdot \ve b = \lambda (\ve a \cdot \ve b).
	\]
	\item 分配律
	\[
		a \cdot (\ve b + \ve c) = \ve a \cdot \ve b + \ve a \cdot \ve c.
	\]
\ITEME
\end{theorem}

\begin{proof}
	自证不难.
\end{proof}

\subsection{内积的应用}

\begin{example}
	用向量证明三角形的余弦定理, 即在 $\triangle ABC$ 中有
	\[
		c^2 = a^2 + b^2 - 2ab \cos C. 
	\]
\end{example}

\begin{solution}
	由向量减法可知 $\Ve{AB} = \Ve{CB} - \Ve{CA}$, 对向量两边同时求与自己的内积可得
	\[
		\begin{aligned}
			(\Ve{AB})^2 &= (\Ve{CB} - \Ve{CA})^2\\
			&= (\Ve{CB})^2 + (\Ve{CA})^2 - 2 (\Ve{CB} \cdot  \Ve{CA})\\
			&= \lvert\Ve{CB}\rvert^2 + \lvert\Ve{CA}\rvert^2 - 2 \lvert\Ve{CB}\rvert  \lvert\Ve{CA}\rvert \cos \langle \Ve{CB},\Ve{CA} \rangle,
		\end{aligned}
	\]
	显然其中 $\lvert\Ve{AB}\rvert = c$, $\lvert\Ve{CA}\rvert = b$, $\lvert\Ve{CB}\rvert = a$, $\langle \Ve{CB},\Ve{CA} \rangle = \cos C$ 则有
	\[
		c^2 = a^2 + b^2 - 2ab \cos C. 
	\]
\end{solution}

\begin{example}
	证明 Schwarz 不等式
	\[
		\lvert \ve a \cdot \ve b \rvert \le \lvert \ve a \rvert \lvert \ve b \rvert.
	\]
\end{example}

\begin{solution}
	由内积定义可知
	\[
		\lvert \ve a \cdot \ve b \rvert = \bigl\lvert \lvert\ve a\rvert \lvert \ve b \rvert \cos\langle \ve a, \ve b \rangle \bigr\rvert,
	\]
	显然 $-1 \le \cos\langle \ve a, \ve b \rangle \le 1$, 则有
	\[
		\lvert \ve a \cdot \ve b \rvert \le \lvert \ve a \rvert \lvert \ve b \rvert.
	\]
\end{solution}

\subsection{外积的定义}

\begin{definition}
	三维向量 $\ve a$ 与 $\ve b$ 的外积 (Vector product) 也是一个三维向量, 记为 $\vtimes{a}{b}$, 所以也称为向量积或叉乘, 其方向垂直于 $\ve a$ 与 $\ve b$ 所构成的平面, 且 $\{\ve a, \ve b, \vtimes{a}{b}\}$ 是右手系, 其大小为
	\[
		\lvert \vtimes{a}{b} \rvert = \lvert \ve a \rvert \lvert \ve b \rvert \sin \langle \ve a, \ve b \rangle.
	\]
	若用坐标表示两个向量则有
	\[
		\ve a\times\ve b=
		\begin{vmatrix}
		\ve i & \ve j & \ve k\\
		a_1 & a_2 & a_3\\
		b_1 & b_2 & b_3
		\end{vmatrix}.
	\]
\end{definition}

显然外积可以用于计算平行四边形的面积, 设有以 $\ve a, \ve b$ 为邻边的平面四边形, 则有
\[
	S = \lvert \ve a \rvert \lvert \ve b \rvert \sin \langle \ve a, \ve b \rangle = \lvert \vtimes{a}{b} \rvert.
\]
这就是外积的模的几何意义.

类似于内积可以判断两向量是否垂直, 外积可以判断两向量是否共线.

\begin{theorem}{外积与共线}{}
	两个非零向量共线的充要条件是其外积为零向量, 即
	\[
		\ve a \parallelsum \ve b \quad \iff \quad \ve a\times\ve b = \ve 0.
	\]
\end{theorem}
也可以用外积计算两向量夹角的正弦
\[
	\sin \langle \ve a, \ve b \rangle = \frac{\lvert \vtimes{a}{b} \rvert}{\lvert \ve a \rvert \lvert \ve b \rvert}.
\]

外积也可以计算与两个向量 $\ve a, \ve b$ 垂直的向量, 则 $\lambda (\vtimes{a}{b})$ 可以表示任意垂直于 $\ve a, \ve b$ 的向量.
\smallskip

求点 $P$ 到直线 $l$ 的距离 $d$, 可以在直线 $l$ 上, 任取一向量 $\Ve{AB}$, 则由外积的模的几何意义, 可得
\[
	d = \frac{\lvert \Ve{AP} \times \Ve{AB} \rvert}{\lvert\Ve{AB}\rvert}.
\]


\subsection{外积的性质}


\begin{lemma}{}{opl1}
设 $\ve a$ 在与 $\ve c$ 垂直的平面 $\alpha$ 上的投影向量为 $\ve a'$, 则
\[
\ve a \times \ve c = \ve a' \times \ve c.
\] 	
\end{lemma}

\begin{proof}
	显然 $\langle \ve a',\ve c \rangle = \frac{\pi}{2}$ 则可得
	\[
		\lvert \ve a \times \ve c\rvert = \lvert \ve a \rvert \lvert \ve c \rvert \sin \langle \ve a,\ve c \rangle,
	\]
	\[
		\lvert \ve a' \times \ve c\rvert = \lvert \ve a' \rvert \lvert \ve c \rvert.
	\]
	又因为 $\langle \ve a',\ve c \rangle = \langle \ve a,\ve c \rangle + \langle \ve a',\ve a \rangle$ 可得
	\[
		\sin \langle \ve a,\ve c \rangle  = \cos \langle \ve a',\ve a \rangle.
	\]
	综上可得
	\[
		\lvert \ve a \times \ve c\rvert = \lvert \ve a \rvert \lvert \ve c \rvert \cos \langle \ve a,\ve a' \rangle = \lvert \ve a' \rvert \lvert \ve c \rvert = \lvert \ve a' \times \ve c\rvert.
	\]
\end{proof}

\begin{lemma}{}{opl2}
	设有向量 $\ve a',\ve c$, 其中 $\lvert \ve c \rvert = 1$, $\ve a'$ 是与 $\ve c$ 垂直的平面 $\alpha$ 上的一个向量, 则将 $\ve a'$ 在 $\alpha$ 上顺时针 (从 $\ve c$ 所指的方向看, 或沿 $\ve c$ 的方向的逆时针) 旋转 $\frac{\pi}{2}$ 得到 $\ve a_1$,则有
	\[
		\ve a_1 = \ve a' \times \ve c.
	\]
\end{lemma}

\begin{proof}
	由定义显然.
\end{proof}


\begin{theorem}{外积运算律} 
对于任意向量 $\ve a, \ve b, \ve c$ 及任意实数 $\lambda$, 有
\ITEMS
	\item 反交换律
	\[
		\vtimes{a}{b} = -\ve b \times \ve a.
	\]
	\item 与数乘的结合律
	\[
		\lambda \vtimes{a}{b} = \lambda (\vtimes{a}{b}).
	\]
	\item 分配律
	\[
		(\ve a + \ve b) \times \ve c = \ve a \times \ve c + \ve b \times \ve c.
	\]
\ITEME
\end{theorem}

\begin{proof}
反交换律与结合律由定义显然, 这里只证明分配律.

不失一般性的我们可以设 $\ve c$ 为单位向量, 则对于与 $\ve c$ 垂直的平面 $\alpha$, 向其作向量 $\ve a, \ve b$ 的投影 $\ve a', \ve b'$ , 由 \hyperref[lem:opl1]{引理 \ref{lem:opl1}} 可知
\[
	\ve a \times \ve c = \ve a' \times \ve c, \quad \ve b \times \ve c = \ve b' \times \ve c,
\]
由 \hyperref[lem:opl2]{引理 \ref{lem:opl2}}, 则将 $a', b'$ 在 $\alpha$ 平面上顺时针旋转 $\frac{\pi}{2}$ , 过程如 \hyperref[fig:ope1]{下图} 所示, 则显然有
\[
	(\ve a + \ve b) \times \ve c = \ve a \times \ve c + \ve b \times \ve c.
\] 
\end{proof}

\begin{example}
用向量证明正弦定理, 即在 $\triangle ABC$ 中有
	\[
		\frac{a}{\sin A} = \frac{b}{\sin B} = \frac{c}{\sin C}.
	\]
\end{example}

\begin{solution}
	设 $\Ve AB = \ve c, \Ve CA = \ve b, \Ve CB = \ve a$ 则由三角形法则显然 $\ve a = \ve b + \ve c$ 则有
	\[
		\vtimes{a}{b} = (\ve b + \ve c) \times \ve b = \ve b \times \ve b + \ve c \times \ve b = \ve c \times \ve b,
	\]
	\[
		\ve c \times \ve a = \ve c \times (\ve b + \ve c) = \ve c \times \ve b + \ve c \times \ve c = \ve c \times \ve b.
	\]
	即
	\[
		\vtimes{a}{b} = \ve c \times \ve b = \ve c \times \ve a.
	\]
	则可知
	\[
		\lvert \ve a \rvert \lvert \ve b \rvert \sin \langle \ve a, \ve b \rangle = \lvert \ve c \rvert \lvert \ve b \rvert \sin \langle \ve c, \ve b \rangle = \lvert \ve c \rvert \lvert \ve a \rvert \sin \langle \ve c, \ve a \rangle,
	\]
	连等式同除以 $\lvert \ve a \rvert \lvert \ve b \rvert \lvert \ve c \rvert$ 可得
	\[
		 \frac{\sin \langle \ve a, \ve b \rangle}{\lvert \ve c \rvert} = \frac{\sin \langle \ve c, \ve b \rangle}{\lvert \ve a \rvert} = \frac{\sin \langle \ve c, \ve a \rangle}{\lvert \ve b \rvert},
	\]
	稍作整理则有
	\[
		\frac{a}{\sin A} = \frac{b}{\sin B} = \frac{c}{\sin C}.
	\]
\end{solution}

\begin{example}
	对于不共线的两个非零向量 $\ve a, \ve b$ 证明
	\[
		(\vtimes{a}{b})^2 = \ve a^2 \ve b^2 - (\ve a \cdot \ve b)^2.
	\]
	并进一步推出三角形 $\triangle ABC$ 的 Heron 公式, 其中 $s$ 为三角形周长的一半.
	\[
		S_{\triangle ABC} = \sqrt{s (s - a)(s - b)(s - c)}.
	\]
\end{example}

\begin{solution}
	由外积与内积的定义可知
	\[
		(\vtimes{a}{b})^2 = \ve a^2 \ve b^2 \sin^2 \langle \ve a, \ve b \rangle,
	\]
	\[
		(\ve a \cdot \ve b)^2 = \ve a^2 \ve b^2 \cos^2 \langle \ve a, \ve b \rangle.
	\]
	则显然有
	\[
		(\vtimes{a}{b})^2 = \ve a^2 \ve b^2 - (\ve a \cdot \ve b)^2.
	\]
	\smallskip
	
	对于三角形 $\triangle ABC$ 有 $\Ve{BC} = \ve a, \Ve{CA} = \ve b, \Ve{AB} = \ve c$ 显然有
	\[
	\begin{aligned}
		\ve a + \ve b + \ve c &= \ve 0\\
		\ve a + \ve b &= -\ve c\\
		(\ve a + \ve b)^2 &= \ve c^2\\
		\ve a^2 + \ve b^2 + 2(\ve a \cdot \ve b) &= \ve c^2\\
		\ve a \cdot \ve b &= \frac{1}{2}\left(\ve c^2 - \ve a^2 - \ve b^2 \right)
	\end{aligned}
	\]
	
	由外积的几何意义有
	\[
		S_{\triangle ABC} = \frac{1}{2} \lvert \vtimes{a}{b} \rvert,
	\]
	综上有
	\[
		\begin{aligned}
			(S_{\triangle ABC})^2 &= \frac{1}{4} \lvert \vtimes{a}{b} \rvert^2\\
			&= \frac{1}{4} \left(a^2 b^2 - (\ve a \cdot \ve b)^2\right)\\
			&= \frac{1}{4} \left(a^2 b^2 - \frac{1}{4}\left(\left(c^2 - a^2 - b^2 \right)^2\right)\right)\\
			&= \frac{1}{16} \left(4 a^2 b^2 - \left(\left(c^2 - a^2 - b^2 \right)^2\right)\right)\\
			&= \frac{1}{16} \left(2ab + \left(c^2 - a^2 - b^2 \right)\right) \left(2ab - \left(c^2 - a^2 - b^2 \right)\right)\\
			&= \frac{1}{16} \left(c^2 - (a - b)^2\right) \left((a + b)^2 - c^2\right)\\
			&=\frac{1}{16} \left(c - a + b\right) \left(c + a - b\right) \left(a + b - c\right) \left(a + b + c\right).
		\end{aligned}
	\]
	代入 $s = a + b + c$ 可得
	\[
		S_{\triangle ABC} = \sqrt{s (s - a)(s - b)(s-c)}.
	\]
\end{solution}

\subsection{向量的混合积}

\begin{definition}{混合积}
我们定义向量 $\ve a, \ve b, \ve c$ 的混合积 (Scalar triple product) 为, 向量 $\ve a$ 与 $\ve b$ 的外积, 再与 $\ve c$ 作内积. 混合积的结果是一个数量, 记为 $(\ve a,\ve b, \ve c)$, 即
\[
	(\ve a,\ve b, \ve c) = (\vtimes{a}{b}) \cdot \ve c.
\]
\end{definition}

同样的我们来分析混合积的几何意义, 首先由定义可知
\[
\begin{aligned}
	(\ve a,\ve b,\ve c) &= (\vtimes{a}{b}) \cdot \ve c\\
	&= \lvert \vtimes{a}{b} \rvert \lvert \ve c \rvert \cos \langle \vtimes{a}{b},\ve c \rangle\\
	&= \lvert \vtimes{a}{b} \rvert \rm{Prj}_{\vtimes{a}{b}} \ve c.
\end{aligned}
\]
则混合积的符号 取决于投影 $\rm{Prj}_{\vtimes{a}{b}} \ve c$ 的符号. 即当 $\ve a, \ve b, \ve c$ 构成右手系时 (如 \hyperref[fig:ltriplep]{图 \ref*{fig:ltriplep}}) 混合积为正, 当 $\ve a, \ve b, \ve c$ 构成左手系时 (如 \hyperref[fig:rtriplep]{图 \ref*{fig:ltriplep}}) 混合积为负.

而且不难发现投影 $\rm{Prj}_{\lvert \vtimes{a}{b} \rvert} \ve c$ 的绝对值即该平行六面体在由 $\ve, a \ve b$ 构成的面上的高, 所以混合积的绝对值是以 $\ve a, \ve b, \ve c$ 为邻边的平行六面体的体积.

由混合积的定义及其几何意义可得
\[
	(\ve a, \ve a , \ve c) = 0,
\]
\[
	(\ve a, \ve b, \ve c) = (\ve c, \ve a, \ve b) = (\ve b, \ve c , \ve a) = - (\ve a, \ve c, \ve b) = - (\ve b, \ve a , \ve c) = - (\ve c, \ve b , \ve a)
\]
\[
	(\ve a_1 + \ve a_2, \ve b, \ve c) = (\ve a_1, \ve b, \ve c) + (\ve a_2, \ve b, \ve c),
\]
\[
	(\lambda\ve a, \ve b, \ve c) = \lambda(\ve a, \ve b, \ve c).
\]

\begin{theorem}{混合积与共面}
	三个向量 $\ve a, \ve b, \ve c$ 共面的充要条件为 $(\ve a, \ve b, \ve c) = 0$
\end{theorem}
\begin{proof}
	我们先证明必要性, 由 \hyperref[def:coplanarity]{共面的定义} 可知存在实数 $\lambda,\mu$ 使得
	\[
		\ve a = \lambda\ve b + \mu\ve c.
	\]
	代入混合积则有
	\[
		(\ve a, \ve b, \ve c) = (\lambda\ve b + \mu\ve c, \ve b, \ve c) = \lambda(\ve b, \ve b, \ve c) + \mu(\ve c, \ve b, \ve c) = 0.
	\]
	
	接下来证明充分性, 当 $(\ve a, \ve b, \ve c) = 0$ 时, 显然有 $\vtimes{a}{b}= \ve 0$ 或 $\ve c = \ve 0$ 或 $\vtimes{a}{b} \perp \ve c$, 其中前两种情况显然可知三向量共面, 而第三种情况由 $\vtimes{a}{b} \perp \ve a$ 和 $\vtimes{a}{b} \perp \ve b$ 可知三向量共面.
\end{proof}

\begin{example}
	设 $\ve a, \ve b, \ve c$ 为三个不共面的向量, 将任意向量 $\ve d$ 分解为向量 $\ve a, \ve b, \ve c$ 的线性组合
	\[
		\ve d = x\ve a + y\ve b + z\ve c.
	\]
\end{example}

\begin{solution}
	注意到
	\[
		(\ve d, \ve b, \ve c) = (x\ve a + y\ve b + z\ve c, \ve b, \ve c) = x(\ve a, \ve b, \ve c) + y(\ve b, \ve b, \ve c) + z(\ve c, \ve b, \ve c) = x(\ve a, \ve b, \ve c).
	\]
	则有
	\[
		x = \frac{(\ve d, \ve b, \ve c)}{(\ve a, \ve b, \ve c)},
	\]
	类似的可得
	\[
		y = \frac{(\ve a, \ve d, \ve c)}{(\ve a, \ve b, \ve c)}, \quad z = \frac{(\ve a, \ve b, \ve d)}{(\ve a, \ve b, \ve c)}.
	\]
\end{solution}

\begin{corollary}
	空间中任一向量关于三个不共面的向量的分解式是唯一的.
\end{corollary}

\subsection{向量的双重外积}

我们现在讨论三个向量 $\ve a, \ve b, \ve c$ 的双重外积 $(\vtimes ab) \times\ve c$, 我们先从特殊一点的情形入手.

\begin{example}
	证明
	\[
		(\vtimes ab) \times \ve a = \ve a^2 \ve b - (\ve a \cdot \ve b) \ve a.
	\]
\end{example}

\begin{solution}
	若 $\ve a, \ve b$ 有一个为零, 则结果显然成立. 若 $\ve a, \ve b$ 非零, 当 $\ve a, \ve b$ 共线时, 设 $\ve a = \lambda \ve b$, 此时等式两边均取零, 结果成立.
	
	当 $\ve a, \ve b$ 不共线时, $\vtimes ab \neq \ve 0$, 根据外积的定义有
	\[
		(\vtimes ab)\times \ve a \perp \vtimes ab,\quad \ve a \perp \vtimes ab,\quad \ve b \perp \vtimes ab.
	\]
	则显然 $(\vtimes ab)\times \ve a$ 显然与 $\ve a, \ve b$ 共面, 由 \hyperref[def:coplanarity]{共面的定义} 有实数 $\lambda, \mu$ 使得
	\[
		(\vtimes ab)\times \ve a = \lambda \ve a + \mu \ve b.
	\]
	我们先设法确定 $\mu$ 的值, 对等式两边同时求与 $(\vtimes ab)\times \ve a$ 的内积.
	\[
	\begin{aligned}
		[(\vtimes ab)\times \ve a] ^2 &= (\lambda \ve a + \mu \ve b) \cdot [(\vtimes ab)\times \ve a]\\
		\lvert(\vtimes ab)\times \ve a\rvert ^2 &= \mu \ve b \cdot [(\vtimes ab)\times \ve a]\\
		\lvert (\vtimes ab) \rvert ^2 \lvert \ve a \rvert^2 \sin^2\langle \vtimes ab, \ve a \rangle &= (\vtimes ab, \ve a, \mu \ve b)\\
		(\vtimes ab)^2 \ve a ^2 &= (\ve a, \mu \ve b, \vtimes ab)\\
		(\vtimes ab)^2 \ve a ^2 &= \mu (\vtimes ab) \cdot (\vtimes ab)\\
		\ve a ^2 &= \mu.
	\end{aligned}
	\]
	
	接下来试着确定 $\lambda$ 的值, 对等式两边同时求与 $\ve a$ 的内积.
	\[
	\begin{aligned}
		[(\vtimes ab) \times \ve a] \cdot \ve a &= (\lambda \ve a + \mu \ve b) \cdot \ve a\\
		0 &= \lambda \ve a ^2 + \ve a^2 \ve b \cdot \ve a\\
		0 &= \lambda + \ve a \cdot \ve b\\
		\lambda &= - \ve a \cdot \ve b.
	\end{aligned}
	\]
	
	综上可得
	\[
		(\vtimes ab)\times \ve a = \ve a^2 \ve b - (\ve a \cdot \ve b) \ve a.
	\]
\end{solution}


\begin{theorem}
	对任意向量 $\ve a, \ve b, \ve c$ 有
	\[
		(\vtimes ab) \times \ve c = (\ve a \cdot \ve c)\ve {b} - (\ve b \cdot \ve c) \ve a.
	\]
\end{theorem}

\begin{proof}
	若 $\ve a, \ve b, \ve c$ 中有一个为零, 则结果显然成立. 若 $\ve a, \ve b, \ve c$ 非零, 且当 $\ve a, \ve b$ 共线时, 有 $\vtimes ab = \ve 0$, 此时等式两边均取零, 结果成立.
	
	若 $\ve a, \ve b$ 不共线, 则 $\vtimes ab \neq \ve 0$, 由外积定义有
	\[
		(\vtimes ab) \times \ve c \perp (\vtimes ab), \quad \ve a \perp (\ve a \times \ve b),  \quad\ve b \perp (\vtimes ab).
	\]
	
	所以 $(\vtimes ab) \times \ve c$ 与 $\ve a, \ve b$ 共面, 由 \hyperref[def:coplanarity]{共面的定义} 有实数 $\lambda, \mu$ 使得
	\[
		(\vtimes ab)\times \ve c = \lambda \ve a + \mu \ve b.
	\]
	这次我们先设法确定 $\lambda$ 的值, 对等式两边同时求与 $\vtimes bc$ 的内积.
	\[
	\begin{aligned}
		(\vtimes bc) \cdot [(\vtimes ab)\times \ve c] &= (\vtimes bc) \cdot (\lambda \ve a + \mu \ve b)\\
		(\vtimes ab, \ve c, \vtimes bc) &= (\ve b, \ve c, \lambda \ve a + \mu \ve b)\\
		(\ve c, \vtimes bc, \vtimes ab) &= (\lambda \ve a + \mu \ve b, \ve b, \ve c)\\
		[\ve c \times (\vtimes bc)] \cdot (\vtimes ab) &= [(\lambda \ve a + \mu \ve b) \times \ve b] \cdot \ve c\\
		[- (\vtimes bc) \times \ve c] \cdot (\vtimes ab) &= \lambda (\ve a,\ve b, \ve c)\\
		[(\vtimes cb) \times \ve c] \cdot (\vtimes ab) &= \lambda (\ve a,\ve b, \ve c)\\
		[\ve c^2 \ve b - (\ve b \cdot \ve c) \ve c] \cdot (\vtimes ab) &= \lambda (\ve a,\ve b, \ve c)\\
		-(\ve b \cdot \ve c) (\ve a,\ve b, \ve c) &= \lambda (\ve a,\ve b, \ve c)\\
		-(\ve b \cdot \ve c) &= \lambda.
	\end{aligned}
	\]
	接下来确定 $\mu$ 的值, 对等式两边同时求与 $\vtimes ca$ 的内积.
	\[
	\begin{aligned}
		(\vtimes ca) \cdot [(\vtimes ab)\times \ve c] &= (\vtimes ca) \cdot (\lambda \ve a + \mu \ve b)\\
		(\vtimes ab, \ve c, \vtimes ca) &= (\ve c, \ve a, \lambda \ve a + \mu \ve b)\\
		(\ve c, \vtimes ca, \vtimes ab) &= (\ve a,\lambda \ve a + \mu \ve b, \ve c)\\
		[\ve c \times (\vtimes ca)] \cdot (\vtimes ab) &= [\ve a \times (\lambda \ve a + \mu \ve b)] \cdot \ve c\\
		[- (\vtimes ca) \times \ve c] \cdot (\vtimes ab) &= \mu (\ve a,\ve b, \ve c)\\
		-[\ve c^2 \ve a - (\ve a \cdot \ve c) \ve c] \cdot (\vtimes ab) &= \mu(\ve a,\ve b, \ve c)\\
		(\ve a \cdot \ve c) (\ve a,\ve b, \ve c) &= \mu (\ve a,\ve b, \ve c)\\
		\ve a \cdot \ve c &= \mu.
	\end{aligned}
	\]
	综上可得
	\[
		(\vtimes ab) \times \ve c = (\ve a \cdot \ve c)\ve {b} - (\ve b \cdot \ve c) \ve a.
	\]
\end{proof}

\begin{corollary}
	\[
	\ve a \times (\vtimes bc) = (\ve a \cdot \ve c) \ve b - (\ve a \cdot \ve b) \ve c.
	\]
\end{corollary}

\begin{corollary}
	\[

\section{随机过程基本概念}

随机过程过程是概率空间 $(\Omega, \mathcal{F}, P)$ 上的一族随机变量 $\{X(t), t\in T\}$, 其中 $T$ 称为指标集或参数集.当指标集取 $\R, \R_+$ 或 $[a, b]$ 时, 称 $\{X(t), t\in T\}$ 为连续参数的随机过程, 当指标集取 $\Z, \Z_+$ 时, 称 $\{X(t), t\in T\}$ 为离散参数的随机过程.若指标集 $T = \{0, 1, 2, \cdots\}$ 时, 称之为随机序列或时间序列.在这里举几个随机过程的例子.
\begin{example}[随机游动]
	一个醉汉在路上行走, 以概率 $p$ 前进一步, 以概率 $1-p$ 后退一步(假定其步长相同).以 $X(t)$ 记 $t$ 时刻他在路上的位置, 则 $\{X(t)\}$ 就是直线上的随机游动.
\end{example}

\begin{example}[布朗运动]
	英国植物学家布朗注意到漂浮在液面上的微小粒子不断进行无规则的运动, 这种运动后来称为布朗运动.它是分子大量随机碰撞的结果, 若以 $(X(t), Y(t))$ 表示粒子在平面坐标上的位置, 则它是平面上的布朗运动.
\end{example}

\begin{example}[排队模型]
	顾客来到服务站要求服务.当服务站中的服务员都忙碌, 即服务员都在为别的顾客服务时, 来到的顾客就要排队等候顾客的到来, 每个顾客所需的服务时间都是随机的, 所以如果用 $X(t)$ 表示 $t$ 时刻的队长, 用 $Y(t)$ 表示 $t$ 时刻到来的顾客所需的等待时间, 则 $\{X(t), t\in T\}$,  $\{Y(t), t\in T\}$ 都是随机过程.
\end{example}

\section{有限维分布与柯尔莫哥洛夫定理}

为了描述随机过程 $\{X(t), t\in T\}$ 的统计特征, 对于每个 $t\in T$, 要知道 $X(t)$ 的分布函数 $F(t, x) = P\{X(t)\le x\}$ (我们称 $F(t, x)$ 为随机过程的一维分布), 以及它们在任意 $n$ 个时间点的联合分, 即随机过程的 $n$ 维分布
\[
	F_{t_1, t_2, \cdots, t_n}(x_1, x_2, \cdots, x_n) = P\{X(t_1) \le x_1, X_(t_2)\le x_2, \cdots, X(t_n)\le x_n\}
\]
随机过程的所有一维分布, 二维分布 $\cdots$ $n$ 维分布等的全体
\[
	\{F_{t_1, t_2, \cdots, t_n}(x_1, x_2, \cdots, x_n), t_1, t_2, \cdots, t_n\in T, n\ge1\}
\]
称为随机过程 $\{X(t), t\in T\}$ 的有限维分布族.
